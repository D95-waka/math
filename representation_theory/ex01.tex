\question{}
יהי $n \ge 2$ ונבחן את ההצגה ה־$n$־מימדית של $S_n$,
\[
	\rho : S_n \to \GL_n(\CC)
\]
המתקבלת על־ידי מיפוי של פרמוטציות $\sigma$ למטריצות פרמוטציה מתאימות.

\subquestion{}
נראה ש־$\rho$ מתפרקת לסכום של שתי הצגות $V_1 \oplus V_2$ כך ש־$\dim V_1 = 1$ וכן $\dim V_2 = n - 1$.
\begin{proof}
	כל מטריצת פרמוטציה $M$ מקיימת את התכונה שסכום כל שורה הוא $1$ וכן סכום כל עמודה הוא $1$, לכן זוהי מטריצה סטוכסטית ומקיימת את התכונה ש־$1$ ערך עצמי של המטריצה עבור הווקטור $v = {(1, \ldots, 1)}^t$.
	נגדיר את $V_1 = \Sp\{ v \}$ ולכן נקבל ש־$\rho(\sigma)(u) = u$ לכל $u \in V_1$ ולכל $\sigma \in S_n$.
	נגדיר את $V_2 = \{ (t_1, \ldots, t_n) \in \CC^n \mid t_1 + \cdots + t_n = 0 \}$, ברור כי $V_1 + V_2 = \CC^n$ שכן $(1, 0, \ldots, 0, -1) \in V_2$ ולכן $(1, 0, \ldots, 0) \in V_1 + V_2$ וכן הלאה.
	יהי $v \in V_1 \cap V_2$, אז $v = \alpha (1, \ldots, 1)$ שכן $v \in V_1$, אבל גם $n \cdot \alpha = 0$ ולכן $\alpha = 0$ בלבד, כלומר $V_1 \cap V_2 = \{ 0 \}$ ונובע ש־$V_1 \oplus V_2 = \CC^n$. \\
	מהגדרת $V_1$ מתקיים $\dim V_1 = 1$, לכן ממשפט המימד נסיק ש־$\dim V_2 = n - 1$.

	אם $\sigma \in S_n$ וכן $(t_1, \ldots, t_n) \in V_2$ אז $\sigma . (t_1, \ldots, t_n) = (t_{\sigma(1)}, \ldots, t_{\sigma(n)})$ ובכל מקרה סכומם $0$ ומתקיים $\sigma . (t_1, \ldots, t_n) \in V_2$.
	אם כך שני המרחבים אינווריאנטיים.
\end{proof}

\subquestion{}
נראה ש־$V_1$ ו־$V_2$ אי־פריקים.
\begin{proof}
	ברור כי $V_1$ אי־פריק, נראה זאת עבור $V_2$. \\
	נגדיר את $u_i \in \CC$ כך ש־$\pi_i(u_i) = 1, \pi_n(u_i) = -1$, לכן $V_2 = \Sp\{u_1, \ldots, u_n\}$.
	כדי להראות ש־$V_2$ אי־פריק מספיק להראות שאפשר לשלוח כל $u_i$ לכל $u_j$ עבור $1 \le i, j \le n$.
	נבחין כי $(i\ j) \in S_n$ וכן $(i\ j) . u_i = u_j$, אז נובע בדיוק שלא יתכן שיש תת־מרחב ממש אינווריאנטי כך שהוא כולל חלק מווקטורי הבסיס $v_1, \ldots, v_n$.
\end{proof}

\subquestion{}
נוכיח שכל הצגה $(V, \rho)$ אי־פריקה של חבורה אבלית סופית $A$ מעל $\CC$ היא בהכרח ממימד $1$.
\begin{proof}
	נניח בשלילה ש־$\dim V \ge 2$ ולכן קיימים $u, v \in V \setminus \{ 0 \}$ בלתי־תלויים לינארית.
	ההצגה היא אי־פריקה ולכן קיים $\sigma \in A$ כך ש־$\sigma . u = v$, או לצירוף לינארי של הווקטורים ולכן נניח בלי הגבלת הכלליות את השוויון בלבד.
	נתון כי $A$ סופית ולכן קיים $n \in \NN$ כך ש־$\sigma^n = e$, בפרט $\sigma^n . u = u$ ונסיק שההעתקה $\rho(\sigma)$ היא נילפוטנטית ולכן אלכסונית, אבל זו סתירה ל־$\sigma . u = v$ שכן הווקטורים יהיו תלויים לינארית.
	זאת אפשר לראות על־ידי הנחה בלי הגבלת הכלליות ש־$u, v$ נבחרו כך שההעתקה $\rho(\sigma)$ היא בצורת ז'ורדן.
\end{proof}

\subquestion{}
נראה שזה לא בהכרח המצב עבור שדות אחרים.
\begin{proof}
	נגדיר את ההצגה $(\RR^2, \rho)$ מעל $\RR$ של החבורה $C_4 = \ZZ / 4 \ZZ$ הצגת הסיבוב, כלומר $C_4 = \langle \sigma \rangle$ עבור,
	\[
		\rho(\sigma)
		= \begin{pmatrix}
			0 & -1 \\
			1 & 0
		\end{pmatrix}
	\]
	נבחין כי בהרצאה כבר ראינו כי העתקה זו היא אי־פריקה.
\end{proof}

\question{}
\begin{definition}[הצגה רגולרית]
	ההצגה הרגולרית (השמאלית) של חבורה סופית $G$ היא הצגה $|G|$־מימדית מעל $\CC$ המוגדרת על־ידי הפעולה של $G$ על עצמה על־יד כפל (משמאל).
	במקרה זה נסמן $(V_{\reg}, \rho_{\reg})$ עבור $V_{\reg}$ מרחב לינארי מעל $\CC$ עם $\dim V_{\reg} = |G|$ והבסיס ${\{ v_g \}}_{g \in G}$ כך שמתקיים,
	\[
		g . v_h
		= \rho_{\reg}(g) \cdot v_h
		= v_{g h}
	\]
\end{definition}

\subquestion{}
נמצא עותק של ההצגה החד־מימדית הטריוויאלית של $G$ ב־$V_{\reg}$.
\begin{solution}
	נגדיר $u = \sum_{g \in G} v_g$ וכן $U = \Sp\{ u \}$ ונראה שהוא מקיים את הטענה.
	יהי $g \in G$,
	\[
		g . u
		= g . (\sum_{h \in G} v_h)
		= \sum_{h \in G} g . v_h
		= \sum_{h \in G} v_{g . h}
		= \sum_{h \in G} v_h
		= u
	\]
	ולכן $g$ פועל טריוויאלית על $u$ והטענה מתקיימת.
\end{solution}

\subquestion{}
נמצא עותק חד־מימדי של הצגת הסימן ב־$S_3$ וההצגה הרגולרית שלה.
\begin{solution}
	נרצה למצוא וקטור $u \in \CC^3$ המקיים,
	\[
		g . u
		= \begin{cases}
			u & g \in S_3 \setminus A_3 \\
			-u & g \in A_3
		\end{cases}
	\]
	נוכל להשיג זאת על־ידי הווקטור $u = \sum_{g \in S_3} \varepsilon_g g$, כאשר $\varepsilon_g \in \{ \pm \}$ מקיים,
	\[
		\varepsilon_g = -1
		\iff g \in A_3
	\]
	מהגדרת $A_3$ נקבל שאם $g \in A_3$ אז $\det \rho(g) = -1$ וכן $g . u = -u$ כפרמוטציה של הרכיבים.
\end{solution}

\subquestion{}
נראה שההצגה הרגולרית של $S_3$ לא מכילה תת־הצגה שאיזומורפית לאחת משתי ההצגות מהסעיפים הקודמים מעבר לאלו שתיארנו.
\begin{proof}
	תהי תת־הצגה $V \subseteq \CC^3$ של $S_3$.
	אם $\dim V \ne 1$ אז סיימנו, לכן נניח ש־$\dim V = 1$ וכן $V = \Sp\{ v \}$.

	נניח ש־$S_3$ פועלת טריוויאלית על $v$ ונראה שבהכרח $v = u$ עבור $u = {(1, 1, 1)}^t$ מסעיף א'.
	למעשה נוכל לקבל זאת על־ידי בחינת הפרמוטציות $(1\ 2), (2\ 3)$, מהן נובע ישירות שהקורדינטות השונות חייבות להיות זהות.

	ההוכחה עבור המקרה של $A_3$ זהה.
\end{proof}

\question{}
נגדיר שתי הצגות דו־מימדיות של $S_3$ על־ידי,
\[
	\rho_1
	= \begin{cases}
		(1\ 2) \mapsto
		\begin{pmatrix}
			-1 & -1 \\
			0 & 1
		\end{pmatrix} \\
		(2\ 3) \mapsto
		\begin{pmatrix}
			0 & 1 \\
			1 & 0
		\end{pmatrix}
	\end{cases}
	\qquad
	\rho_2
	= \begin{cases}
		(1\ 2) \mapsto
		\begin{pmatrix}
			0 & \omega \\
			\omega^2 & 0
		\end{pmatrix} \\
		(2\ 3) \mapsto
		\begin{pmatrix}
			0 & 1 \\
			1 & 0
		\end{pmatrix}
	\end{cases}
\]
כאשר $\omega \in \CC$ שורש יחידה פרימיטיבי מסדר $3$.

\subquestion{}
נסביר למה אלה הגדרות להצגות תקינות.
\begin{solution}
	תוצאה ידועה מתורת החבורות היא ש־$S_n$ נוצר על־ידי שתי פרמוטציות מתאימות,
	במקרה של $S_3$ זוהי בחירת פרמוטציות הולמת.
\end{solution}

\subquestion{}
נראה ששתי ההצגות הללו איזומורפיות.
\begin{proof}
	נרצה לבנות איזומורפיזם בין ההצגות, כלומר לבנות $T : \CC^2 \to \CC^2$ המקיימת,
	\[
		\rho_2 \circ T
		= T \circ \rho_1
	\]
	נבחין כי מספיק להגדיר את ההעתקה על בסיס של $\CC^2$ כהומומורפיזם לינארי.
	ננסה למצוא אחת, נגדיר,
	\[
		T
		= \begin{pmatrix}
			a & b \\
			c & d
		\end{pmatrix}
	\]
	אז צריך להתקיים מחילופיות,
	\[
		\begin{pmatrix}
			0 & 1 \\
			1 & 0
		\end{pmatrix} T
		= T
		\begin{pmatrix}
			0 & 1 \\
			1 & 0
		\end{pmatrix}
		\implies
		\begin{pmatrix}
			c & d \\
			a & b
		\end{pmatrix}
		= \begin{pmatrix}
			b & a \\
			d & c
		\end{pmatrix}
	\]
	ונובע ש־$c = b, a = d$, כלומר,
	\[
		T
		= \begin{pmatrix}
			a & b \\
			b & a
		\end{pmatrix}
	\]
	נשתמש שוב בחילופיות עבור $(1\ 2)$ ונקבל,
	\[
		\begin{pmatrix}
			-a - b & -b - a \\
			b & a
		\end{pmatrix}
		= \begin{pmatrix}
			-1 & -1 \\
			0 & 1
		\end{pmatrix} T
		= T \begin{pmatrix}
			0 & \omega \\
			\omega^2 & 0
		\end{pmatrix}
		= \begin{pmatrix}
			b \omega^2 & a \omega \\
			a \omega^2 & b \omega
		\end{pmatrix}
	\]
	בפרט נקבל $a = b \omega$ ולכן נוכל להגדיר $b = 1, a = \omega$ ונקבל שההעתקה,
	\[
		T
		= \begin{pmatrix}
			\omega & 1 \\
			1 & \omega
		\end{pmatrix}
	\]
	מקיימת את הטענה ומעידה על האיזומורפיזם.
\end{proof}

\subquestion{}
תהי הצגה $(V, \rho)$ של חבורה $G$ ויהי תת־מרחב אינווריאנטי $U \le V$.
נגדיר את הצגת המנה על $V / U$ ונראה כי זו אכן הצגה.
\begin{proof}
	נסמן את $\pi : V \to V / U$ ההטלה של מרחב המנה, כלומר $\pi(v) = v + U$ לכל $v \in V$.
	נגדיר את ההצגה $\bar{\rho} : G \to \GL(V / U)$ על־ידי,
	\[
		(\bar{\rho}(g))(v + U)
		= (\pi \circ \rho(g))(v)
	\]
	לכל $v + U \in V / U$. \\
	עלינו להראות שפונקציה זו מוגדרת היטב וכן שזהו הומומורפיזם חבורות.

	למעשה ההגדרה של אינווריאנטיות שקולה לנורמליות של תת־מרחב לינארי כתת־חבורה, לכן ממשפט האיזומורפיזם הראשון נקבל את שני התנאים.
\end{proof}

\subquestion{}
נוכיח שלכל הצגה ממימד סופי $(V, \rho)$ יש סדרת הרכב,
כלומר סדרה סופית של תת־מרחבים אינווריאנטיים,
\[
	\{ 0 \} = V_0 \le V_1 \le \cdots \le V_k = V
\]
כך שהצגת המנה $V_i / V_{i - 1}$ כולן אי־פריקות.
\begin{proof}
	נסמן $\Aa = \{ U < V \mid U \text{ is invariant} \}$, נבחין כי $\{ 0 \} \in \Aa$ ולכן בפרט $\Aa$ לא ריק.
	אם $\deg V = n$ אז $\deg \Aa \subseteq [n]$ ולכן קיים מספר מקסימלי $0 \le m < n$ כך שקיים מרחב לינארי ממימד $m$ ב־$\Aa$, אז יהי $U \in \Aa$ מרחב כזה.
	מסעיף א' נובע שהדרישה עבור המנה חלה.
	נמשיך בהליך זה, מובטח מבנייתו שבכל חזרה המימד ירד ולכן מסופיות מימד $V$ נובע שהתהליך יסתיים ונקבל שרשרת כדרוש.
\end{proof}
