\question{}
יהי $R$ חוג חילופי ויהיו $M_1, M_2, N$ מודולים מעל $R$.
נראה שמתקיים,
\[
	(M_1 \oplus M_2) \otimes_R N
	\cong (M_1 \otimes_R N) \oplus (M_2 \otimes_R N)
\]
\begin{proof}
	נגדיר את ההעתקה,
	\[
		\varphi : (M_1 \oplus M_2) \times N \to (M_1 \otimes_R N) \oplus (M_2 \otimes_R N),
		\qquad
		\varphi((m_1, m_2), n)
		= ((m_1 \otimes n), (m_2 \otimes n))
	\]
	נבחין כי העתקה זו מוגדרת היטב מהגדרת החיבור הישר החיצוני כמכפלה קרטזית עם הגדרת פעולת כפל.
	מאותה הסיבה זוהי גם העתקה $R$־בילינארית.
	\begin{multline*}
		\varphi((m_1, m_2) + (m_1', m_2'), n)
		= \varphi((m_1 + m_1', m_2 + m_2'), n)
		= ((m_1 + m_1') \otimes n, (m_2 + m_2') \otimes n) \\
		= (m_1 \otimes n + m_1' \otimes n, m_2 \otimes n + m_2' \otimes n)
		= (m_1 \otimes n, m_2 \otimes n) + (m_1' \otimes n, m_2' \otimes n)
	\end{multline*}
	ונוכל באופן זהה לקבל את כל התכונות שמעידות על בילינאריות על־ידי שימוש בבילינאריות של מכפלה טנזורית. \\
	מהתכונה האוניברסלית קיימת הרמה,
	\[
		\tilde{\varphi} : (M_1 \oplus M_2) \otimes_R N \to (M_1 \otimes_R N) \oplus (M_2 \otimes_R N)
	\]
	זהו הומומורפיזם, ונרצה להראות שהוא הפיך.
	נגדיר את ההעתקה,
	\[
		\psi : (M_1 \otimes_R N) \oplus (M_2 \otimes_R N) \to (M_1 \oplus M_2) \otimes_R N,
		\qquad
		\psi(m_1 \otimes n, m_2 \otimes n')
		= (m_1, 0) \otimes n + (0, m_2) \otimes n'
	\]
	על ידי שימוש בהרמות של $M_1 \times N \to (M_1 \oplus M_2), m \mapsto (m_1, 0) \otimes n$ והעתקה מקבילה לציר השני,
	נקבל שהעתקה זו אכן מוגדרת היטב, ואף שהיא הומומורפיזם מודולים, ונבדוק שהוא הפוך ל־$\tilde{\varphi}$.
	\begin{align*}
		\tilde{\varphi}(\psi(m_1 \otimes n, m_2 \otimes n'))
		& = \tilde{\varphi}((m_1, 0) \otimes n + (0, m_2) \otimes n') \\
		& = \tilde{\varphi}((m_1, 0) \otimes n) + \tilde{\varphi}((0, m_2) \otimes n') \\
		& = \varphi((m_1, 0), n) + \varphi((0, m_2), n') \\
		& = (m_1 \otimes n, 0 \otimes n) + (0 \otimes n', m_2 \otimes n') \\
		& = (m_1 \otimes n, m_2 \otimes n')
	\end{align*}
	ומהצד השני,
	\[
		\psi(\tilde{\varphi}((m_1, m_2) \otimes n))
		= \psi(\varphi((m_1, m_2), n))
		= \psi((m_1 \otimes n), (m_2 \otimes n))
		= (m_1, 0) \otimes n + (0, m_2) \otimes n
		= (m_1, m_2) \otimes n
	\]
	ומצאנו שאכן $\tilde{\varphi}$ איזומורפיזם המעיד על הרצוי.
\end{proof}

נמצא שקילות פשוטה לחבורות האבליות הבאות.

\subquestion{}
נגדיר $A = \Hom_{\ZZ}(\ZZ / m \ZZ, \ZZ / n \ZZ)$ עבור $m, n \in \NN$.
\begin{solution}
	יהי $f : \ZZ / m \ZZ \to \ZZ / n \ZZ$ הומומורפיזם, כלומר $f \in A$, ונסמן $k + n \ZZ = f(\bar{1})$ עבור $0 \le k < n$.
	מתקיים $f(\bar{x}) = f(x \cdot \bar{1}) = x f(\bar{1}) = x k + n \ZZ$, נסיק ש־$f$ נקבע ביחידות על־ידי $k$.
	נבחין גם כי אם $f(x) = k x$ וכן $g(x) = l x$ אז $(f + g)(x) = (k + l) x$, ולכן נראה שמתקיים $A \cong \ZZ / s \ZZ$ עבור איזשהו $s \in \NN$.
	מתקיים,
	\[
		f(m)
		= f(0)
		= 0
	\]
	וכן,
	\[
		f(n)
		= k n
		= 0
	\]
	ולכן נסיק ש־$s \mid l, k$, ובפרט נובע ש־$s = \gcd(n, m)$ בלבד, נראה שמתקיים $A \cong \ZZ / \gcd(m, n) \ZZ$.

	נגדיר את $\varphi : \ZZ / \gcd(m, m) \ZZ \to A$ על־ידי $\varphi(k)(x) = k x$.
	נובע מהבדיקה שעשינו קודם שמתקיים $(\varphi(k) + \varphi(l))(x) = (k + l)(x) + n \ZZ = k x + l x + n \ZZ = (\varphi(k) + \varphi(l))(x)$, וכן $(a \varphi(k))(x) = x + a k + n \ZZ = \varphi(a k)$.
	לכן $\varphi$ אכן הומומורפיזם, ונשאר להראות שהוא חד־חד ערכי ועל. \\
	נניח ש־$f = \varphi(k) \in \ker \varphi$, כלומר $f(x) = 0$ לכל $x$, בפרט נובע $x k \in n \ZZ$ או $x k \in m \ZZ$ לכל $x$,
	אז גם $k \mid m, n$ על־ידי הצבת $x = 1$, אבל $k < \gcd(m, n)$ ולכן $k = 0$ בלבד, לכן קיבלנו חד־חד ערכיות.
	לכל $f(x) = k x + n \ZZ$ נוכל להניח בלי הגבלת הכלליות ש־$0 \le k < n$, אבל גם $0 \le k < m$ אחרת נקבל $k x = (k - m) x$, בפרט $k \nmid m, n$, כלומר נוכל לבחור $f = \varphi(k)$ עבור $0 \le k < \gcd(m, n)$.
	נסיק ש־$\varphi$ על ולכן איזומורפיזם.
\end{solution}

\subquestion{}
נגדיר $A = \ZZ / m \ZZ \otimes_{\ZZ} \ZZ / n \ZZ$ ל־$m, n \in \NN$.
\begin{solution}
	נבחין כי לכל $b \in \ZZ / n \ZZ$ מתקיים,
	\[
		n \otimes b
		= 1 \otimes (n b)
		= 1 \otimes 0
		= 0
	\]
	ובאופן דומה נקבל גם $a \otimes m = 0$,
	ננחש ונראה שמתקיים $A \cong \ZZ / \gcd(n, m) \ZZ$.

	נגדיר את $\varphi : \ZZ / m \ZZ \times \ZZ / n \ZZ \to \ZZ / \gcd(m, n) \ZZ$ על־ידי $\varphi(a, b) \equiv a b$, זוהי העתקה בילינארית כמכפלה, ולכן קיימת הרמה יחידה,
	\[
		\tilde{\varphi} : \ZZ / m \ZZ \otimes \ZZ / n \ZZ \to \ZZ / \gcd(m, n) \ZZ,
		\qquad
		\tilde{\varphi}(a \otimes b) \equiv a b
	\]
	מהצד השני נגדיר את ההעתקה $\psi : \ZZ / \gcd(n, m) \ZZ \to A$ על־ידי $\psi(a) = a \otimes 1$, ברור כי העתקה זו מוגדרת היטב, וברור כי זהו הומומורפיזם על־ידי בילינאריות של מכפלה טנזורית.

	מבילינאריות $a \otimes b = (a b) \otimes 1$,
	נשתמש בתכונה זו כדי לראות ששני ההומומורפיזמים הפוכים,
	\[
		\psi(\tilde{\varphi}(a \otimes b))
		= \psi(\varphi(a, b))
		= \psi(a b)
		= a b \otimes 1
		= a \otimes b
	\]
	ומהצד השני,
	\[
		\tilde{\varphi}(\psi(a))
		= \tilde{\varphi}(a \otimes 1)
		= \varphi(a, 1)
		= a \cdot 1
		= a
	\]
	כלומר קיבלנו שהם אכן הפוכים ולכן מעידים על האיזומורפיזם $A \cong \ZZ / \gcd(m, n) \ZZ$
\end{solution}

\subquestion{}
נגדיר את $A = \QQ \otimes_{\ZZ} \QQ$.
\begin{solution}
	יהי $\frac{p}{q} \otimes \frac{a}{b} \in A$ טנזור פשוט, אז מבילינאריות מתקיים,
	\[
		\frac{p}{q} \otimes \frac{a}{b}
		= \frac{b p}{b q} \otimes \frac{a}{b}
		= \frac{p}{b q} \otimes \frac{a b}{b}
		= \frac{p}{b q} \otimes a
		= \frac{a p}{b q} \otimes 1
	\]
	נשתמש בתכונה זו כדי להראות ש־$A \cong \QQ$.
	נגדיר את ההעתקה $\varphi : \QQ \times \QQ \to \QQ$ על־ידי $\varphi(p, q) = p q$, זוהי העתקה בילינארית כמכפלה ולכן ניתן להרים אותה ל־$\tilde{\varphi} : A \to \QQ$ על־ידי,
	\[
		\tilde{\varphi}(p \otimes q)
		= p q
	\]
	ומהצד השני שוב נגדיר את $\psi : \QQ \to \QQ \otimes_{\QQ} \QQ$ על־ידי $\psi(q) = q \otimes 1$, וברור כי זהו הומומורפיזם על־ידי לינאריות ברכיב הראשון.
	נקבל כך,
	\[
		\tilde{\varphi}(\psi(q))
		= \varphi(q, 1)
		= q
	\]
	וכן,
	\[
		\psi(\tilde{\varphi}(p \otimes q))
		= \psi(p q)
		= p q \otimes 1
		= p \otimes q
	\]
	וקיבלנו ש־$A \cong \QQ$ כפי ששני האיזומורפיזמים מעידים.
\end{solution}

\question{}
נראה שכל מכפלה טנזורית מעל חוג חילופי $R$ היא אסוציאטיבית, כלומר נראה שאם $M, N, P$ מודולים מעל $R$ אז מתקיים,
\[
	(M \otimes_R N) \otimes_R P
	\cong M \otimes_R (N \otimes_R P)
\]
\begin{proof}
	נשתמש באסטרטגיה הסטנדרטית של הגדרת הומומורפיזמים הפוכים והרמתם. \\
	נקבע $p \in P$ ונגדיר את ההעתקה $f_p : M \times N \to M \otimes_R (N \otimes_R P)$ על־ידי,
	\[
		f_p(m, n)
		= m \otimes (n \otimes p)
	\]
	אז ברור כי היא מוגדרת היטב, והיא בילינארית על־ידי בילינאריות המכפלה הטנזורית, לכן ניתן להרים את $f_p$ להומומורפיזם $\tilde{f}_p : M \otimes_R N \to M \otimes_R (N \otimes_R P)$,
	ומתקיים,
	\[
		\tilde{f}_p(m \otimes n)
		= m \otimes (n \otimes p)
	\]
	עתה נוכל להגדיר את ההעתקה הכללית,
	\[
		\varphi : (M \otimes_R N) \times P \to M \otimes_R (N \otimes_R P)
		\qquad
		\varphi(m \otimes n, p)
		= \tilde{f}_p(m \otimes n)
		= m \otimes (n \otimes p)
	\]
	אז ברור כי היא מוגדרת היטב וכן שהיא בילינארית ולכן ניתנת להרמה להומומורפיזם,
	\[
		\tilde{\varphi} : (M \otimes_R N) \otimes_R P \to M \otimes_R (N \otimes_R P)
	\]
	על־ידי הגדרת והרמת פונקציה דומה ל־$f_p$ עבור הקורדינטה השנייה נוכל להסיק שגם,
	\[
		\psi : M \times (N \otimes_R P) \to (M \otimes_R N) \otimes_R P
		\qquad
		\psi(m, n \otimes p)
		= (m \otimes n) \otimes p
	\]
	ניתנת להרמה להומומורפיזם,
	\[
		\tilde{\psi} : M \otimes_R (N \otimes_R P) \to (M \otimes_R N) \otimes_R P
		\qquad
		\tilde{\psi}(m \otimes (n \otimes p))
		= (m \otimes n) \otimes p
	\]
	על־ידי הצבה ישירה נקבל ש־$\tilde{\varphi}, \tilde{\psi}$ הומומורפיזמים הפוכים ולכן איזומורפיזמים המעידים על הטענה.
\end{proof}

\question{}
יהיו $U, V$ מרחבים וקטוריים מעל שדה $\FF$ עם הבסיסים ${\{ u_i \}}_{i \in I}, {\{ v_j \}}_{j \in J}$ בהתאמה. \\
נראה ש־${\{ u_i \times v_j \}}_{i \in I, j \in J}$ הוא בסיס עבור המכפלה הטנזורית $U \otimes_{\FF} V$.
\begin{proof}
	יהי $u \otimes v \in U \otimes_{\FF} V$ טנזור פשוט, לכן קיימים צירופים סופיים כך שמתקיים,
	\[
		u = \sum_i \alpha_i u_i,
		\qquad
		v = \sum_j \beta_j v_j
	\]
	כאיברים במרחב וקטורי עם בסיס נתון.
	מבילינאריות נקבל,
	\[
		u \otimes v
		= \left(\sum_i \alpha_i u_i\right) \otimes \left(\sum_j \beta_j v_j\right)
		= \sum_{i, j} \alpha_i \beta_j (u_i \otimes v_j)
	\]
	כלומר מצאנו שכל טנזור פשוט ניתן לייצוג כצירוף לינארי סופי. הטענה עבור טנזורים לא פשוטים היא זהה, שכן כל טנזור כזה הוא סכום סופי של טנזורים פשוטים, וסכום סופי של סכומים סופיים הוא סופי ולכן צירוף לינארי.
	נסיק אם כך שהקבוצה ${\{ u_i \otimes v_j \}}_{i \in I, j \in J}$ היא קבוצה פורשת של $U \otimes_{\FF} V$.
	נשאר להראות שפירוק זה הוא יחיד, אבל תוצאה זו מתקבלת ישירות מהעובדה ש־$u_i, v_j$ בסיסים ולכן עבורם יש פירוק יחיד, וזהו למעשה הפירוק שהצגנו עבור הטנזורים הפשוטים.
\end{proof}

\question{}
יהי $R$ חוג כלשהו, ונזכור כי מודול מעל $R$ נקרא חופשי אם יש לו בסיס, נראה שלוש הגדרות שונות עבור בסיס:
\begin{definition}[בסיס כמקיים תכונת הרמה אוניברסלית]
	בסיס של $M$ הוא תת־קבוצה $S \subseteq M$ המקיימת את התכונה האוניברסלית, כלומר שלכל העתקה $f : S \to L$ לאיזשהו $R$־מודול $L$ יש הרחבה יחידה להומומורפיזם $\hat{f} : M \to L$.
\end{definition}
\begin{definition}[בסיס כקבוצה יוצרת וייחודית]
	בסיס של $M$ הוא תת־קבוצה $S \subseteq M$ כך שכל איבר ב־$M$ ניתן לייצוג כצירוף לינארי $\sum_{s \in S} r_s . s$ בצורה יחידה, וכאשר $r_s = 0$ כמעט תמיד.
\end{definition}
\begin{definition}[בסיס כקבוצה לא תלויה־לינארית]
	בסיס של $M$ הוא תת־קבוצה $S \subseteq M$ היוצרת את $M$ כך שאם $\sum_{s \in S} r_s . s = 0$ אז $r_s = 0$ לכל $s \in S$.
\end{definition}

\subquestion{}
נראה שכל שלוש ההגדרות שקולות.
\begin{proof}
	$1 \implies 2$:
	נגדיר את $N \subseteq R[S]$ על־ידי $N = \Sp_R\{ x_s \}$, כאשר $R[S]$ פולינומים במשתנים מתאימים ל־$S$, ונסמן $x_s$ לכל $s \in S$ המשתנה המתאים לקריאות.
	תהי $f : S \to N$, כלומר העתקה ששולחת את ערכי $M$ לצירופים לינאריים של $\{ x_s \mid s \in S \}$.
	נגדיר את $f(s) = x_s$ לכל $s \in S$, לכן קיים הומומורפיזם יחיד $\bar{f} : M \to N$.

	נבחין כי לכל $p = \sum_{s \in S} r_s x_s \in N$ נוכל לקבל $\bar{f}(\sum r_s s) = \sum r_s \bar{f}(s) = \sum r_s f(s) = p$, לכן $\bar{f}$ על. \\
	בנוסף אם $p = \sum p_s x_s, q = \sum q_s x_s$ כך ש־$\bar{f}(p) = \bar{f}(q)$ אז נקבל,
	\[
		0
		= \bar{f}(p) - \bar{f}(q)
		= \bar{f}(p - q)
		= \sum_{s \in S} (p_s - q_s) x_s
	\]
	ונסיק שבהכרח $p_s = q_s$, כלומר $p = q$, ונובע ש־$\bar{f}$ חד־חד ערכי ולכן איזומורפיזם.

	יהי $m \in M$ כלשהו, ונניח שגם $\bar{f}(m) = \sum_{s \in S} r_s x_s$.
	מתקיים אם כך,
	\[
		\bar{f}(m)
		= \sum_{s \in S} r_s x_s
		= \sum_{s \in S} r_s \bar{f}(s)
		= \bar{f}(\sum_{s \in S} r_s s)
	\]
	ולכן משימוש ב־$\bar{f}^{-1}$ על שני האגפים,
	\[
		m
		= \sum_{s \in S} r_s s
	\]
	כלומר מצאנו כי קיים צירוף לינארי של $S$ ל־$m$.

	נניח שמתקיים גם $\bar{f}(m) = \sum_{s \in S} t_s . s$, כלומר שקיימת עוד דרך לקבל את $m$ כצירוף לינארי של איברי $S$, אז נקבל גם,
	\[
		\sum_{s \in S} r_s x_s
		= \sum_{s \in S} t_s x_s
		\implies
		\sum_{s \in S} (r_s - t_s) x_s
		= 0
	\]
	כלומר נובע $r_s = t_s$ לכל $s \in S$, בהתאם מצאנו כי קיים ייצוג יחיד כצירוף לינארי.

	$2 \implies 3$:
	הנחנו שכל איבר ניתן לייצוג כצירוף לינארי יחיד, לכן נסמן $0 = \sum_s r_s . s$ הצירוף היחיד המקיים זאת.
	נניח כי קיים $r_s \ne 0$, אז מתקיים $-r_s . s = \sum_{t \in S \setminus \{ s \}} r_t . s$, אבל נזכור כי גם ל־$-r_s . s$ יש ייצוג יחיד, והוא כמובן $-r_s . s$, לכן $r_t = 0$ לכל $t \ne s$, כלומר $-r_s . s = 0$.
	נבחין כי אם $r_s$ היה מחלק אפס אז היינו מקבלים שלכל ייצוג שבו $s$ לא מופיע, כלומר $\langle S \setminus \{ s \} \rangle$ ניתן להוסיף את $r_s s$, וכך יש שני ייצוגים כצירוף לינארי, לכן $r_s$ לא מחלק אפס.
	הנחנו ש־$r_s \ne 0$ ולכן נובע ש־$s = 0$ בלבד, אבל אם $0 \in S$ אז נקבל שלא יכול להיות שלכל איבר יהיה צירוף לינארי מתאים יחיד, ולכן נקבל סתירה, כלומר $r_s = 0$ לכל $s \in S$, בדיוק הגדרת בסיס כקבוצה לא תלויה לינארית.

	$3 \implies 2$:
	יהי $m \in M$ ונניח ש־$m = \sum_s r_s . s = \sum_s t_s . s$, אז בפרט נקבל $\sum_s (r_s - t_s) . s = 0$, אבל אז נובע ש־$r_s - t_s = 0$, כלומר $r_s = t_s$, ונקבל שאכן קיים ייצוג יחיד כצירוף לינארי.

	$2 \implies 1$:
	תהי העתקה $f : S \to L$ עבור $L$ מודול כלשהו מעל $R$, ונרצה להראות שיש הרחבה יחידה להומומורפיזם $\hat{f} : M \to L$.
	נגדיר את ההעתקה $\hat{f} : M \to L$ על־ידי,
	\[
		\hat{f}(\sum_{s \in S} r_s s)
		= \sum_{s \in S} r_s f(s)
	\]
	פונקציה זו מוגדרת היטב שכן לכל $m \in M$ יש צירוף לינארי יחיד כזה, ובהתאם הגדרה יחידה בהצגה כסכום.
	מתקיים גם,
	\[
		r \bar{f}(m)
		= r \bar{f}(\sum r_s s)
		= r \sum r_s f(s)
		= \sum (r r_s) f(s)
		= \bar{f}(r m)
	\]
	ואם גם $n \in M$ כך ש־$n = \sum t_s s$ אז,
	\[
		\bar{f}(m) + \bar{f}(n)
		= \sum r_s f(s) + \sum t_s f(s)
		= \sum (r_s + t_s) f(s)
		= \bar{f}(\sum (r_s + t_s) s)
		= \bar{f}(m + n)
	\]
	ונסיק ש־$\bar{f}$ הומומורפיזם, כלומר קיימת הרחבה של $f$ לאיזשהו הומומורפיזם.

	תהי איזושהי הרחבה $\hat{f} : M \to L$ להומומורפיזם.
	נסמן $m = \sum_s r_s . s$ הייצוג היחיד של $m$ כצירוף לינארי מעל $S$, אז מתקיים,
	\[
		\hat{f}(m)
		= \sum_{s \in S} r_s . \hat{f}(s)
		= \sum_{s \in S} r_s . f(s)
		= \bar{f}(m)
	\]
	כלומר מצאנו כי $\hat{f}$ נקבע ביחידות על־ידי $f$ ובפרט מזדהה עם $\bar{f}$, ולכן קיימת הרחבה יחידה להומומורפיזם.
\end{proof}

\subquestion{}
נראה שאם $M_1$ ו־$M_2$ הם מודולים חופשיים עם בסיסים $S_1$ ו־$S_2$ בהתאמה, וכן ש־$|S_1| = |S_2|$, אז בהכרח $M_1 \cong M_2$.
\begin{proof}
	מהנתון $|S_1| = |S_2|$ נובע שקיימת פונקציה חד־חד ערכית ועל $h : S_1 \to S_2$. \\
	נגדיר את ההעתקה $f : S_1 \to M_2$ על־ידי $f(s_1) = h(s_1)$, אז קיימת הרחבה יחידה $F : M_1 \to M_2$ להומומורפיזם כך ש־$F(s_1) = h(s_1)$ לכל $s_1 \in S_1$.

	נבדוק ש־$F$ חד־חד ערכית, יהי $m_1 \in M$ כך ש־$m_1 \in \ker F$.
	אם $m_1 = \sum_{s_1 \in S_1} r_{s_1} s_1$ אז נסיק,
	\[
		0
		= F(m_1)
		= \sum_{s_1 \in S_1} r_{s_1} F(s_1)
		= \sum_{s_2 \in S_2} r_{h^{-1}(s_2)} s_2
	\]
	כאשר המעברים מוצדקים על־ידי העובדה ש־$r_{s_1} = 0$ כמעט תמיד.
	כלומר קיבלנו צירוף לינארי של איברי $S_2$ המתאפס, לכן $r_{s_1} = 0$ לכל $s_1 \in S_1$, בפרט $m_1 = 0$ ונקבל שהגרעין טריוויאלי ו־$F$ חד־חד ערכית.

	נרצה להראות ש־$F$ גם על, יהי $m_2 \in M_2$, אז הוא מתפרק ל־$m_2 = \sum_{s_2} r_{s_2} . s_2$ יחיד, אבל,
	\[
		m_2
		= \sum_{s_2 \in S_2} r_{s_2} . s_2
		= \sum_{s_1 \in S_1} r_{h(s_1)} . h(s_1)
		= \sum_{s_1 \in S_1} r_{h(s_1)} . F(s_1)
		= F\left(\sum_{s_1 \in S_1} r_{h(s_1)} . s_1\right)
	\]
	כלומר מצאנו מקור ל־$m_2$ ולכן $F$ היא גם על, ונסיק כי זהו איזומורפיזם $F : M_1 \to M_2$.
\end{proof}

\question{}
יהי חוג $R$, ונגדיר את $R^n = \bigoplus_{[n]} R$, זהו מודול על $R$.

\subquestion{}
נראה ש־$R^n$ הוא מודול חופשי.
\begin{proof}
	נבחן את הסכום הישר כסכום ישר חיצוני, כלומר נבחן את $R^n$ כ־$\prod_{[n]} R$ עם הפעולה של כפל כלל האיברים.
	בשל כך נזהה את $R^n$ הן כמכפלה קרטזית והן כסכום ישר והפעולה.
	נגדיר גם את $e_i(k) = 1$ אם $i = k$ ו־$e_i(k) = 0$ אחרת, ונקבל ש־$E = \{ e_i \mid i \in [n] \}$ הוא קבוצה יוצרת של $R^n$.
	לבסוף, אם $\bar{0} = \sum_{i \in [n]} r_i . e_i$ אז בפרט $0 = \pi_i(\bar{0}) = r_i$ ונסיק מהשאלה הקודמת שזהו בסיס.
\end{proof}

\subquestion{}
יהי $I \trianglelefteq R$ אידאל, נסמן את תת־הקבוצה של $R^n$,
\[
	I R^n
	= \{ i . \bar{r} \mid i \in I, \bar{r} \in R^n \}
\]
וכן נגדיר את $I^n = \bigoplus_{[n]} I$, ונראה ש־$\langle I R^n \rangle = I^n$, כלומר תת־המודול הנוצר על־ידי $I R^n$ הוא בדיוק $I^n$.
\begin{proof}
	יהי $i . \bar{r} \in I R^n$, ונסמן $r_k = \pi_k(\bar{r})$ ההטלה של $R^n$, אז $i r_k \in I$ לכל $k$, אבל $i . \bar{r} = (i r_1, \ldots, i r_n)$, לכן גם $i . \bar{r} \in I^n$.
	נסיק ש־$I R^n \subseteq I^n$, וכקבוצת יוצרים ובשל סגירות של $I^n$ לחיבור, נסיק שגם $\langle I R^n \rangle \subseteq I^n$.

	בכיוון ההפוך יהי $\bar{\imath} \in I^n$ ונסמן $\bar{\imath} = (i_1, \ldots, i_n)$.
	אז מתקיים,
	\[
		\bar{\imath}
		= \sum_{k = 1}^n i_k . e_k
	\]
	אבל $i_k \in I, e_k \in R^n$ ולכן $i_k . e_k \in I R^n$, ומסגירות של תת־מודול לסכומים סופיים נסיק שגם $I^n \subseteq \langle I R^n \rangle$.
\end{proof}

\subquestion{}
נראה על־ידי שימוש בלמה של צורן שלכל חוג לא טריוויאלי יש אידאל מקסימלי.
\begin{proof}
	נסמן $S = \{ I \triangleleft R \mid I \ne R \}$ כל האידאלים ב־$R$ ונבדוק את תנאי הלמה של צורן עבור יחס ההכלה.
	תהי $T \subseteq S$ שרשרת ביחס ההכלה וניקח את $I = \{ 0 \} \cup \bigcup T$ ונראה ש־$I \in S$ וכן שהוא חסם עליון של $T$.
	אכן $J \subseteq I$ לכל $J \in T$, ונראה ש־$I \triangleleft R, I \ne R$. \\
	יהיו $a, b \in I$, ונניח בלי הגבלת הכלליות ש־$T$ לא ריקה. קיימים $J, K \in S$ כך ש־$b \in K, a \in J$.
	$T$ שרשרת ולכן $J \subseteq K$ או $K \subseteq J$ ונניח בלי הגבלת הכלליות ש־$K \subseteq J$.
	אז $a, b \in J$ ולכן $a + b \in J \subseteq I$ כנדרש.
	אם $c \in I$ ו־$a \in R$ אז קיים $J \in T$ כך ש־$c \in J$ ולכן גם $a c \in J \subseteq I$ ולכן נוכל להסיק סגירות לכפל משמאל, כפל מימין נובע באופן שקול ולכן נסיק ש־$I \triangleleft R$.
	בנוסף $1 \notin J$ לכל $J \in T$ ולכן גם $1 \notin I$ ובפרט $I \ne R$.
	בהתאם $I \in S$ הוא אכן חסם עליון ולכן תנאי הלמה של צורן מתקיימים וקיים אידאל מקסימלי.
\end{proof}

\subquestion{}
נראה שאם $R$ הוא חוג חילופי, אז $R^m \cong R^n$ אם ורק אם $m = n$. \\
נסיק שלכל מודול חופשי נוצר סופית מעל $R$ יש דרגה מוגדרת היטב.
\begin{proof}
	אם $m = n$ אז נקבל ש־$R^m = R^n$ בדיוק.

	בכיוון ההפוך, נניח שמתקיים $R^m \cong R^n$ ונסמן $\varphi : R^m \to R^n$ איזומורפיזם מודולים המעיד על כך.
	נסמן $I \trianglelefteq R$ אידאל מקסימלי, קיים כזה ממשפט האידאל המקסימלי.
	אם $i \in I, \bar{r} \in R^m$ אז מתקיים $i . \varphi(\bar{r}) = \varphi(i \bar{r})$ ולכן גם,
	\[
		\varphi(I R^m)
		\subseteq I R^n
	\]
	ועל־ידי טענה דומה עם $\varphi^{-1}$ נקבל שוויון.
	מסעיף ב' נסיק בהתאם שמתקיים גם,
	\[
		\varphi(I^m)
		= I^n
	\]
	נגדיר את $\psi : R^m / I R^m \to R^n / I R^n$ על־ידי,
	\[
		\psi(\bar{r} + I R^m)
		= \varphi(\bar{r} + I R^m)
		= \varphi(\bar{r}) + I R^n
	\]
	כאשר השקילות האחרונה נובעת מ־$\varphi(I R^m) = I R^n$. \\
	נבחין גם כי $f : R^m / I R^m \to {(R / I)}^m$ המוגדר על־ידי $f(\bar{r} + I R^m) = (r_1 + I, \ldots, r_m + I)$ הוא הומומורפיזם מודולים, וכן אם נסמן $K = R / I$ אז $K$ שדה ולכן $f : R^m / I R^m \to K^n$.
	נובע גם ש־$f$ איזומורפיזם, הוא חד־חד ערכי ועל מבדיקה ישירה, ולכן נוכל לבחון את ההעתקה $g \circ \psi \circ f^{-1}$ עבור $g$ העתקה דומה ל־$f$ עבור $n$, זהו איזומורפיזם $K^m \to K^n$,
	בפרט זהו איזומורפיזם לינארי ולכן $n = m$ בלבד.

	יהי $R$ חילופי ו־$M$ מודול מעל $R$ חופשי ונוצר סופית, ונסמן $s_1, \ldots, s_n$ הבסיס של $M$.
	אז מתקיים $M \cong R^n$ על־ידי שימוש בהגדרה שקולה לבסיס כהרמה יחידה, ולכן $n$ הוא המספר היחיד שעבורו קיים איזומורפיזם כזה, לכן נוכל לסמן $\rank_R M = n$ וסימון זה יהיה מוגדר היטב.
\end{proof}

\subquestion{}
נבדוק למה טענה זו לא נכונה עבור $R$ לא חילופי.
\begin{solution}
	בהוכחת הטענה השתמשנו בעובדה ש־$R / I$ הוא שדה, טענה זו נכונה רק במקרה שבו $R$ הוא חילופי, במקרה שבו החוג לא חילופי יתכן ש־$R / I$ לא יהיה חוג חילוק.
\end{solution}

\question{}
בשאלה זו נראה שהטענה של השאלה הקודמת אכן לא נכונה לחוגים חילופיים. \\
נגדיר את החבורה האבלית,
\[
	B
	= \prod_{i \in \NN} \ZZ
\]
ונגדיר את $R = \End(B)$.

\subquestion{}
נראה ש־$R$ לא חילופי.
\begin{proof}
	נגדיר את ההעתקות $f, g \in \End(B)$ על־ידי,
	\[
		f(a)(i) = a(i + 1),
		\quad
		g(a)(i) = 2^i \cdot a(i)
	\]
	כלומר ההעתקה הראשונה מזיזה את איברי הסדרה אחורה (ומדלגת על האיבר הראשון), וההעתקה השנייה מכפילה כל איבר בסדרה בשתיים בחזקת מיקומו.
	נבחין כי אנו מזהים את $B$ עם $\ZZ^{\NN}$ מטעמי קריאות.
	בנוסף מתקיים,
	\[
		(f \circ g)(a)(i) = 2^{i + 1} \cdot a(i + 1),
		\qquad
		(g \circ f)(a)(i) = 2^i a(i + 1)
	\]
	כלומר נקבל שתי העתקות שונות.
\end{proof}

\subquestion{}
נגדיר את ההעתקות $\varphi_1, \varphi_2 \in R$ על־ידי,
\[
	\varphi_1(a)(i) = a(2 i - 1),
	\qquad
	\varphi_2(a)(i) = a(2 i),
\]
ונראה שההעתקה $\varphi : R^2 \to R$ המוגדרת על־ידי $\varphi(r_1, r_2) = r_1 \varphi_1 + r_2 \varphi_2$ היא איזומורפיזם.
\begin{proof}
	יהי $s \in R$, אז $s$ הוא הומומורפיזם מודולים ובפרט משמר חיבור ולכן,
	\[
		s . \varphi(r_1, r_2)
		= s (r_1 \varphi_1 + r_2 \varphi_2)
		= s r_1 \varphi_1 + s r_2 \varphi_2
		= \varphi(s r_1, s r_2)
		= \varphi(s . (r_1, r_2))
	\]
	נוכל באופן זהה לחלוטין לקבל ש־$\varphi$ משמר כפל באיברי $R$ ולכן נסיק ש־$\varphi$ הומומורפיזם, נשאר להראות שהוא חד־חד ערכי ועל.

	נראה ש־$\varphi$ על.
	נגדיר את $\psi_1, \psi_2 \in R$ על־ידי,
	\[
		\psi_1(a)(i) = \begin{cases}
			a(k) & \exists k \in \NN,\ i = 2 k - 1 \\
			0 & \text{otherwise}
		\end{cases},
		\qquad
		\psi_2(a)(i) = \begin{cases}
			a(k) & \exists k \in \NN,\ i = 2 k \\
			0 & \text{otherwise}
		\end{cases}
	\]
	כלומר $\psi_1$ שולח סדרה לסדרה של איברים אי־זוגיים ומשאיר אפס במקומות הזוגיים, ו־$\psi_2$ פועל באופן דומה בזוגיים.
	לכן (וגם בבדיקה ישירה) נקבל,
	\[
		\psi_1(\varphi_1(a))(i)
		= \begin{cases}
			a(i) & i \in 2 \NN - 1 \\
			0 & \text{othereise}
		\end{cases},
		\qquad
		\psi_2(\varphi_2(a))(i)
		= \begin{cases}
			a(i) & i \in 2 \NN \\
			0 & \text{othereise}
		\end{cases}
	\]
	בפרט נובע $\id = \psi_1 \varphi_1 + \psi_2 \varphi_2$.
	יהי $a \in R$, ונגדיר את $a_1 = a \psi_1, a_2 = a \psi_2$, אז $a_1, a_2 \in R$ ומתקיים $\varphi(a_1, a_2) = a \psi_1 \varphi_1 + a \psi_2 \varphi_2 = a \id = a$, ונסיק ש־$\varphi$ על.

	נניח ש־$(r_1, r_2) \in \ker \varphi$, כלומר $\varphi(r_1, r_2)(a) = 0$ לכל $a \in B$.
	נסמן $\rho_1 = \psi_1 \varphi_1$.
	בפרט גם $\varphi(r_1, r_2)(\rho_1 a) = 0$ אבל,
	\[
		0
		= \varphi(r_1, r_2)(\rho_1(a))
		= (r_1 \varphi_1 \rho_1)(a) + (r_2 \varphi_1 \rho_1)(a)
		= (r_1 \varphi_1 \psi_1 \varphi_1)(a) + 0
		= r_1(\varphi_1(a))
	\]
	לכל $b \in B$ יש $a$ כך ש־$b = \varphi_1(a)$ ולכן נסיק שמתקיים $0 = r_1(b)$ לכל $b \in B$, כלומר $r_1 = 0$ בלבד.
	באופן שקול נוכל להסיק שגם $r_2 = 0$ בלבד, ולכן מצאנו ש־$\ker \varphi = \{ \bar{0} \}$, כלומר $\varphi$ חד־חד ערכי.

	נובע ש־$\varphi$ איזומורפיזם המעיד על $R^2 \cong R$.
\end{proof}

\subquestion{}
נסיק ש־$R^m \cong R^n$ לכל $n, m \in \NN$.
\begin{proof}
	מספיק להראות את הטענה $R^n \cong R$ לכל $n \in \NN$, במצב זה נוכל להרכיב איזומורפיזמים ולקבל את הטענה.
	נראה את האיזומורפיזם הזה באינדוקציה על $n$.
	עבור $n = 1, 2$ הטענה טריוויאלית ונובעת מהסעיף הקודם.
	נניח ש־$\psi : R^n \to R$ איזומורפיזם ונבנה איזומורפיזם $\rho : R^{n + 1} \to R$.
	נגדיר את $\rho(r_1, \ldots, r_{n + 1}) = \psi(r_1, \ldots, r_{n - 1}, \varphi(r_n, r_{n + 1}))$.
	זהו הומומורפיזם כהרכבת הומומורפיזמים, וכן חד־חד ערכי ועל מאותה הסיבה, ולכן איזומורפיזם כפי שרצינו.
\end{proof}
