\begin{enumerate}
    \item מתחו קו או כתבו את הזוגות המתאימים של הקבוצות השוות:
    \begin{center}
    \begin{tabular}{c c c}
    $\{\}$ & \qquad\qquad & $\{1, 2, 2, 3\}$ \\
    $\{\text{סוס}, \text{כלב}, \text{דג}\}$ & \qquad\qquad & $\{a, b, c\}$ \\
    $\{a, a, b, b, c\}$ & \qquad\qquad & $\emptyset$ \\
    $\{1, 3, 2\}$ & \qquad\qquad & $\{\text{דג}, \text{כלב}, \text{סוס}, \text{סוס}\}$
    \end{tabular}
    \end{center}

    \item נתונה הקבוצה $A = \{1, 2, \{1\}, \{3\}\}$. \\
    כתבו נכון או לא נכון:
	\begin{enumerate}
		\item $1 \in A$
		\item $3 \in A$
		\item $\emptyset \in A$
		\item $\emptyset \subseteq A$
		\item $\{2\} \in A$
		\item $\{2\} \subseteq A$
		\item $\{\{1\}, \{3\}\} \subseteq A$
		\item $\{1, 2, 3\} = A$
	\end{enumerate}

    \item פתרו את התרגילים הבאים:
		\begin{enumerate}
			\item $[-4, 7] \cap [4, 7] =$
			\item $[-4, 7] \cup [4, 7] =$
			\item $(2, 5] \cap [4, 5) =$
			\item $\emptyset \cap (1, 2) =$
			\item $\emptyset \cup (3, 5) =$
			\item $\{3, 4, 7\} \cap \{1, 2, 3\} =$
			\item $\{1, 2, 3\} \setminus \{1, 2, 3, 4\} =$
		\end{enumerate}

    \item כתבו בצורה מילולית את הטענות הבאות:
		\begin{enumerate}
			\item $\forall x \in A \cap B, x \in A \cup B$
			\item $\nexists p \in \mathbb{Q} : p \notin \mathbb{R}$
		\end{enumerate}

	\item ציירו את הפונקציות הבאות וקבעו את התחום הטווח והתמונה שלהן:
		\begin{enumerate}
			\item $f(x) = x$
			\item $f(x) = -2 x + 10$
			\item $-x (x - 3)$
			\item $x^2 + 10$
		\end{enumerate}
\end{enumerate}
