\subsection{Introduction}
Suppose that $V, W$ are finite dimension vector spaces over $\RR$.
Notice that in such spaces all norms are equivalent.
a function $f : U \to W$ where $U \subseteq V$ is differentiable if for all $x \in U$ there exists a linear map $df \mid_x = f'(x) : V \to W$ such that,
\[
	f(x + h)
	= f(x) + (df \mid_x)(h) + o(|h|)
\]
Therefore $f' : U \to \Hom(V, W)$ defined by $x \mapsto f'(x)$.
We continue to define in such manner the next differentials recursively, $f'' : U \to \Hom(V, \Hom(V, W))$ etc.
A function $f$ is $C^\infty$ (smooth) if the derivatives of any order exists and are continuous.

To study manifolds we have to be able to move between manifolds and Euclidean spaces, the map from a local open set in euclidean space into a manifold is called local parameterization or local coordinates.
\begin{definition}[Chart]
	Let $X$ be a topological space.
	\textbf{Chart} on $X$ modeled on $V$, $\varphi : U \to \Omega$ is a homeomorphism such that both $U \subseteq X$ and $\Omega \subseteq V$ are open sets.
\end{definition}
In Adams book chart is defined the other way around ($\Omega_{\alpha} \to X_{\alpha}$), meaning that Adams chart is equivalent to our definition to parameterization.
Compatible charts $\varphi_{\alpha}, \varphi_{\beta}$ are charts such that there is a homeomorphism between them and are invertible.
\begin{definition}[Atlas]
	An atlas on $X$ is a family of compatible charts $\{ \varphi_{\alpha} : X_{\alpha} \to \Omega_{\alpha} \mid \alpha \in I \}$ for some indices set $I$,
	such that $\bigcup_{\alpha \in I} X_{\alpha} = X$, namely a family of charts that form a cover of $X$.
\end{definition}
Each atlas can be enriched to contain any chart of an open subset of $X$, thus making it unique maximal atlas.
Two atlases are equivalent if their union is an atlas, equivalently if they share a maximal atlas.
\begin{exercise}
	Prove that this is an equivalent relation.
\end{exercise}
\begin{definition}[Smooth manifold]
	A (smooth) manifold is a Hausdorff 2nd countable topological space $X$ equipped with an equivalence class of atlases (or a maximal atlas).
\end{definition}
Given manifolds $X, Y$ modeled on vector-spaces $V, W$, $X \times Y$ is a manifold modeled on $V \times W$.
$\{ \varphi_{\alpha} : X_{\alpha} \to \Omega_{\alpha} \}$, $\{ \psi_{\beta} : Y_{\beta} \to \Omega_{\beta}' \}$ are atlases,
then $\{ \varphi_{\alpha} \times \psi_{\beta} : X_{\alpha} \times Y_{\beta} \to \Omega_{\alpha} \times \Omega_{\beta'} \}$ is an atlas in $X \times Y$.
Equivalent atlases on $X$ \& on $Y$  yield equivalent atlases on $X \times Y$.
There exists a unique manifold structure on $X \times Y$ such that $X \times Y \to X$ \& $X \times Y \to Y$ are smooth and for any $Z$ manifold and $Z \to X \times Y$ is smooth iff the compositions are smooth.
\begin{definition}[smooth map between manifolds]
	Given $X, Y$ are manifolds modeled on $V, W$ and atlases $\{ \varphi_{\alpha} \}$ and $\{ \psi_{\beta} \}$.
	We say that $f : X \to Y$ is smooth if $\forall \alpha, \beta,\ \varphi_{\beta} \circ f \circ \psi_{\alpha}^{-1}$ is smooth.
\end{definition}
\begin{definition}[Lie group]
	A Lie group $G$ is a set equipped with:
	\begin{enumerate}
		\item a manifold structure
		\item a group structure:
			a group operation $\mu : G \times G \to G$ defined as $\mu(a, b) = a \cdot b$ and inversion $i : G \to G$ defined as $i(x) = x^{-1}$.
	\end{enumerate}
	such that $\mu$ and $i$ are smooth.
\end{definition}
\begin{definition}[Homomorphism of Lie groups]
	Let $G, H$ be Lie groups, a function $f : G \to H$ that is smooth and a group Homomorphism is called Lie group Homomorphism.
\end{definition}
\begin{example}
	$(\RR^n, +)$ is a Lie group.
\end{example}
\begin{example}
	Observe $(\RR^n / \ZZ^n, +)$, an $n$-dimensional torus, is a Lie group.
	Recall that in general $\RR^n / \ZZ^n \cong {(\RR / \ZZ)}^n$ it suffices to show that $S^1$ is a Lie group.
	$(S^1, +)$ is indeed a manifold, and it is clear that it is a group as well.
\end{example}
\begin{example}
	Let $V$ be a vector-space over $\RR$ or $\CC$, denoted as $\Lambda$, and let $\End(V) = \Hom_{\Lambda}(V, V)$ the set of linear operators over $V$.
	Recall that if $\dim_{\Lambda} V = n$ then $\End(V) \cong \RR^{n \times n}$.
	Let,
	\[
		\Aut(V) = \{ T \in \End(V) \mid \exists S \in \End(V),\ T \circ S = S \circ T = \id \} \subseteq \End(V)
	\]
	be the subset of invertible $\Lambda$-linear maps, then $\Aut(V)$ is an open subset of $\End(V)$. \\
	$a, b \mapsto a \cdot b$ as matrix multiplication is a polynomial, and is in particular smooth.
	It follows from Kramer's rule that $a \mapsto a^{-1}$ the matrix inverse can be obtained by $a^{-1} = \frac{1}{\det a} \operatorname{adj}(a)$, and so it's smooth as well.
\end{example}
\begin{remark}
	For $X, Y$ are manifolds and $f : X \to Y$ the following are equivalent:
	\begin{enumerate}
		\item $f$ is smooth
		\item For any $n \in \NN$ and open subset $S \subseteq \RR^n$ \& smooth map $g : S \to X$, the composition $f \circ g$ is smooth
		\item For any smooth function $g : Y \to \RR$, $f \circ g$ is smooth as well
	\end{enumerate}
\end{remark}

	Prove this
\end{exercise}
\begin{definition}[Tangent space]
	Tangent space to a manifold $X$ at $x \in X$ is defined as,
	\[
		T_x X
		= \{ \gamma : (-1, 1) \to X \mid \gamma \text{ is smooth}, \gamma(0) = x \} / \sim
	\]
	where we define that $\gamma_1 \sim \gamma_2$ iff for any smooth function $f : U \to \RR$ where $x \in U$ is an open neighborhood, 
	\[
		\frac{d}{dt} \mid_{t = 0} f \circ \gamma_1
		= \frac{d}{dt} \mid_{t = 0} f \circ \gamma_2
	\]
	In other words, we identify curves by their ``derivative'' when passing through $x$.
\end{definition}
\begin{example}
	For $x \in X \subseteq V$ open set, namely an open subset of a vector space, $T_x X = V$, as can be seem via $\gamma(t) = x + t v$ for $v \in V$.
\end{example}
Smooth map of manifolds $f : X \to Y$ induces $f' = f_* = df \mid_x : T_x X \to T_{f(x)} Y$. \\

\[
	{(\varphi_{\alpha})}^* : T_x X_{\alpha} \to T_{\varphi_{\alpha}(x)} \Omega_{\alpha} = V
\]

\begin{definition}[Lie algebra]
	Suppose that $G$ is a Lie group and $e \in G$ its unit, we define $L(G) = T_e G$ (as a vector-spaces),
	Lie algebra of $G$.
	Also notated as $\operatorname{Lie}(G)$ or $\mathfrak{G}$.
\end{definition}
Let $G$ be a Lie group, $x \in G$, and define left multiplication,
\[
	L_x : G \to x g
\]
defined a diffeomorphism $G \to G$ by $e \mapsto x$ hence a linear isomorphism $T_e G \xrightarrow{\sim} T_x G$,
then left trivial injection $TG \cong G \times L(G)$.
