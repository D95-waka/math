\section{שיעור 1 --- 18.10.2026}
אלגברה קומוטטיבית היא למעשה התורה של חוגים חילופיים (commutative).
היסטורית האלגברה הקומוטטיבית קיימת עבור תורת המספרים, הגאומטריה האלגברית ותורת האינווריאנטים אליה לא ניכנס.

\subsection{מבוא}
המושג הראשי שעליו נדבר בקורס הוא חוג חילופי (commutative ring), אובייקט הדומה לשדה אך ללא חילוק, ואידאל (ideal), המקבילה של תורת החוגים עבור תת־החבורה בתורת החבורות.
נעסוק גם במודול (module), המקבילה של מרחב וקטורי אך מעל חוג סקלרים במקום שדה סקלרים.

הקורס מבוסס בעיקר על הספר של Atiyah Macdoald שנקרא Introduction to commutative algebra, פרקים 1 עד 10.
\begin{definition}[חוג]
	הוא קבוצה עם שתי פעולות בינאריות, $(A, + : A \times A \to A, \cdot : A \times A \to A)$ המקיים את התכונות:
	\begin{enumerate}
		\item $(A, +)$ הוא חבורה חילופית, כלומר אסוציאטיבית, בעלת איבר נייטרלי (נסמן $0$) ולכל איבר יש איבר נגדי ביחס לפעולה
		\item חוק הקיבוץ בכפל: לכל $a, b, c \in A$ מתקיים $a (b c) = (a b) c$
		\item קיום איבר נייטרלי ביחס לכפל: קיים איבר $1 \in A$ כך ש־$1 \cdot a = a \cdot 1 = a$ לכל $a \in A$
		\item חלוק הפילוג (distributivity): לכל $a, b, c \in A$ מתקיים $(a + b) c = a c + b c$ וכן $a (b + c) = a b + a c$
	\end{enumerate}
\end{definition}
\begin{remark}
	האיבר הנייטרלי ביחס לכפל הוא יחיד, זאת שכן,
	\[
		1'
		= 1 \cdot 1'
		= 1
	\]
\end{remark}
\begin{notation}
	ההופכי ביחס לחיבור יסומן $-a$, ובהתאם $a - b = a + (-b)$.
\end{notation}
\begin{definition}[חוג חילופי]
	חוג $(A, +, \cdot)$ נקרא חילופי (commutative) יש חילופיות גם ביחס לכפל, כלומר,
	\[
		\forall a, b \in A,\ 
		a \cdot b = b \cdot a
	\]
\end{definition}
\begin{remark}
	חילופיות החיבור נובעת משאר התכונות.
	זאת שכן,
	\[
		(a + b) (1 + 1)
		= (a + b) 1 + (a + b) 1
		= a \cdot 1 + b \cdot 1 + a \cdot 1 + b \cdot 1
		= a + b + a + b
	\]
	אבל מהצד השני,
	\[
		(1 + 1) (a + b)
		= a + a + b + b
	\]
	$(A, +)$ היא חבורה ולכן נוכל לצמצם ולקבל בדיוק ש־$a + b = b + a$.
\end{remark}
\begin{remark}
	מקיבוציות הכפל נובע שמכפלה $a_1 \cdot a_k$ אינה תלויה בהצבת סוגריים, ולכן מותר להשמיט את הסוגריים.
\end{remark}
\begin{example}
	חוג השלמים $\ZZ$ עם פעולות הכפל והחיבור הסטנדרטיות הוא חוג חילופי.
\end{example}
\begin{example}
	לכל שדה $\FF$, $(\FF, +, \cdot)$ הוא חוג חילופי גם כן.
\end{example}
\begin{example}
	חוג המטריצות $A = M_n(\FF)$, כלומר המטריצות מעל $\FF$ מגודל $n \times n$ כאשר $1_A = I_n$ ו־$0_A = 0$.
	זהו חוג שחילופי אם ורק אם $n = 1$.
	נבחין גם כי במקום שדה $\FF$ יכולנו לבחור חוג כללי.
\end{example}
\begin{example}
	חוג הפולינומים,
	יהי $\FF$ שדה, נסמן את $\FF[x]$ להיות אוסף הפולינומים במשתנה $x$ עם מקדמים ב־$\FF$, וזהו חוג חילופי.
\end{example}
\begin{example}
	החוג הטריוויאלי:
	יהי $A$ יחידון, אז נגדיר את $+, \cdot$ טריוויאלית ונקבל ש־$(A, +, \cdot)$ הוא חוג חילופי.
	בחוג זה אף מתקיים $1_A = 0_A$, וזוהי תכונה המאפיינת ביחידות חוג זה, כלומר אם $0 = 1$ אז מתקיים,
	\[
		a
		= a (1 + 0)
		= a ( 1 + 1)
		= a + a
	\]
	כלומר $a = 0$ לכל $a \in A$.
\end{example}

\subsection{פעולות בסיסיות בחוגים}
יהי $(A, +, \cdot)$ חוג.
\begin{definition}[חזקה]
	אם $n \in \NN$ ו־$a \in A$ אז נגדיר,
	\[
		a^n
		= \overbrace{a \cdots a}^{n \text{ times}}
	\]
\end{definition}
\begin{proposition}[תכונות חזקה]
	לכל $n, m \in \NN$ ו־$a, b \in A$ מתקיים,
	\begin{enumerate}
		\item $a^{n + m} = a^n a^m$
		\item אם $A$ חילופי אז גם,
			\[
				{(a b)}^n = a^n b^n
			\]
	\end{enumerate}
\end{proposition}
\begin{definition}[הכפלה במספר שלם]
	אם $n \in \NN$ ו־$a \in A$ אז נגדיר את,
	\[
		n a
		= \begin{cases}
			\overbrace{a + \cdots + a}^{n \text{times}} & n > 0 \\
			0 & n = 0 \\
			\overbrace{(-a) + \cdots + (-a)}^{n \text{times}} & n < 0
		\end{cases}
	\]
\end{definition}
לכפל בשלם יש אסוציאטיביות.

\subsection{יחידות}
\begin{definition}[איבר הפיך]
	$a \in A$ נקרא הפיך (invertible) או יחידה (unit) אם קיים $b \in A$ כך שמתקיים $a b = b a = 1$.
	במקרה זה נאמר ש־$b$ הוא ההופכי של $a$ ונסמן $b = a^{-1}$.
\end{definition}
\begin{notation}
	נסמן את קבוצת ההפיכים על־ידי,
	\[
		A^{\times}
		= \{ a \in A \mid a \text{ is invertible} \}
	\]
	סימון מקובל נוסף הוא $A^*$.
\end{notation}
\begin{proposition}
	$A^\times$ חבורה ביחס לכפל.
\end{proposition}
\begin{proof}
	נבדוק ש־$A^\times$ סגור ביחס לכפל.
	אם $a, b \in A^\times$ אז קיימים $a', b' \in A^\times$ המקיימים $a a' = b b' = 1$.
	מתקיים,
	\[
		(a b) (b' a')
		= a b b' a'
		= a a'
		= 1
	\]
	וכן באופן דומה נקבל זאת עבור כפל הפוך, לכן $a b \in A^\times$.
	בהתאם נובע שכפל הוא פעולה בינארית על $A^\times$, וכן ברור כי $1 \in A^\times$ נייטרלי וכן שלכל $a \in A^\times$ גם $a' \in A^\times$.
\end{proof}
לחבורה $A^\times$ קוראים גם החבורה הכפלית של החוג (multiplication group).

\subsection{מחלקי אפס}
\begin{definition}[מחלק אפס]
	יהי $A$ חוג חילופי.
	איבר $a \in A$ נקרא מחלק אפס (zero divisor) אם קיים $0 \ne b \in A$ כך ש־$a b = 0$.
\end{definition}
\begin{remark}
	תמיד מתקיים $a \cdot 0 = 0 \cdot a = 0$ בכל חוג, זאת שכן,
	\[
		a \cdot 0
		= a (1 - 1)
		= a \cdot 1 - a \cdot 1
		= a - a
		= 0
	\]
\end{remark}
\begin{exercise}
	הראו כי בכל חוג $A$ מתקיים,
	\[
		a - a = 0
		\qquad
		a (b - c) = a b - a c
		\qquad
		(a - b) c
		= a c - b c
	\]
	לכל $a, b, c \in A$.
\end{exercise}
\begin{example}
	הראו כי אם $a \in A$ מחלק אפס ו־$b \in A$ איבר כלשהו אז $a b$ מחלק אפס.
\end{example}
\begin{definition}[תחום שלמות]
	חוג חילופי $A$ נקרא תחום שלמות (integral domain) או בקיצור תחום,
	אם $A \ne \{ e \}$ וכן אין ב־$A$ מחלקי אפס.
\end{definition}
נעבור לדוגמות למחלקי אפס.
\begin{example}
	אם $A$ חוג לא טריוויאלי אז $0$ הוא מחלק אפס,
	שכן $1 \cdot  0 = 0$.
\end{example}
\begin{example}
	אם $A = \ZZ / 6 \ZZ$ אז $2, 3$ מחלקי אפס שכן $(2 + 6 \ZZ) \cdot (3 + 6 \ZZ) = 0 + 6 \ZZ$.
\end{example}
\begin{example}
	$\ZZ, \FF, \FF[x]$ לכל שדה $\FF$ הם תחומי שלמות.
\end{example}
אך כמובן $\ZZ / 6 \ZZ$ אינו תחום שלמות.

\subsection{אידאלים}
באלגברה קומוטטיבית נהוג לסמן $A \subset B$ להכלה או שוויון ו־$A \subsetneq B$ עבור הכלה ממש. \\
\begin{notation}
	אם $X, Y \subset A$ תת־קבוצות של חוג $A$ אז נסמן,
	\[
		X \pm Y
		= \{ x \pm y \mid x \in X, y \in Y \},
		\quad
		X \cdot Y
		= \{ x y \mid x \in X, y \in Y \}
	\]
	וכן לכל $a \in A$ גם,
	\[
		a X
		= \{ a x \mid x \in X \},
		\quad
		X a
		= \{ x a \mid x \in X \}
	\]
	ולבסוף $-X = \{ -x \mid x \in X \} = (-1) \cdot X$.
\end{notation}
\begin{definition}[אידאל]
	תת־קבוצה $I \subset A$ תיקרא אידאל (Ideal) אם מתקיים:
	\begin{enumerate}
		\item $I$ תת־חבורה של $A$ ביחס לחיבור
		\item $I A = A I = I$, כלומר לכל $b \in I$ ו־$a \in A$ מתקיים $a b, b a \in I$
	\end{enumerate}
\end{definition}
\begin{notation}
	אם $I \subset A$ אידאל אז נסמן $I \triangleleft A$.
\end{notation}
\begin{remark}
	כדי לבדוק ש־$\emptyset \ne I \subset A$ אידאל מספיק לבדוק ש־$I$ סגורה לחיבור ($I + I \subset I$), \\
	וכן שיש סגירות לכפל באיברי $A$ ($I A, A I \subset I$).
\end{remark}
נבחין כי $0 = 0 \cdot a \in I$ לכל $a \in A$ וכן $-a = (-1) \cdot a \in I$ ולכן $(I, +)$ תת־חבורה. \\
בנוסף $I A = I, A I = I$ שכן $I = I \cdot 1 \subset I A$. \\
נבחין כי $I$ לא בהכרח חוג כי אין לו בהכרח יחידה כפלית.
\begin{example}
	אם $A$ חוג כלשהו אז $I = \{ 0 \}$ הוא אידאל האפס. \\
	אידאל היחידה או האידאל הטריוויאלי הוא $I = A$.
\end{example}
\begin{corollary}
	אידאל הוא טריוויאלי אם ורק אם $1 \in I$.
\end{corollary}
\begin{example}
	נגדיר,
	\[
		A
		= \left\{ \begin{pmatrix} a & b \\ 0 & c \end{pmatrix} \mid a, b, c \in \FF \right\}
	\]
	עבור $\FF$ שדה
	 אז מתקיים,
	 \[
	 	I
		= \left\{ \begin{pmatrix} 0 & b \\ 0 & 0 \end{pmatrix} \mid b \in \FF \right\}
	 \]
	 הוא אידאל של $A$, כלומר $I \triangleleft A$.
\end{example}
\begin{example}
	אם $X$ קבוצה ו־$A = \RR^\times = \{ f : X \to \RR \}$ ונגדיר את הפעולות $(f g)(x) = f(x) g(x), (f + g)(x) = f(x) + g(x)$ אז $A$ הוא חוג. \\
	יהי $x_0 \in X$ ונגדיר,
	\[
		I
		= \{ f \in A \mid f(x_0) = 0 \}
	\]
	אז $I \triangleleft A$.
\end{example}
\begin{example}
	עבור $A = \ZZ$ ולכל $n \in \NN \cup \{ 0 \} = \NN_0$ מתקיים $n \ZZ \triangleleft  \ZZ$.
	מתקיים $n \ZZ + n \ZZ = n \ZZ$ וכן $a (n \ZZ) = (a n) \ZZ \subset n \ZZ$. \\
	בכיוון ההפוך כל $I \triangleleft  \ZZ$ הוא מהצורה הזו כי כל תת־החבורות של $\ZZ$ הן מהצורה הזו.
\end{example}
\begin{definition}[אידאל שמאלי וימני]
	יהי $A$ חוג.
	אידאל שמאלי ב־$A$ הוא תת־קבוצה $I \subset A$ כך ש־$(I, +)$ היא תת־חבורה וכן $A I = I$.
	אידאל ימני הוא זהה אך ש־$I A = I$ במקום.
	כלומר אידאל שמאלי סגור לכפל משמאל ואידאל ימני סגור לכפל מימין.
\end{definition}
אידאל שמאלי שהוא גם ימני נקרא לעתים אידאל דו־צדדי. בחוגים חילופיים כל ההגדרות הללו שקולות.

נעבור לפעולות על אידאלים.
\begin{proposition}
	יהי $A$ חוג ויהיו $I_{\alpha} \triangleleft  A$ אוסף אידאלים עבור $\alpha \in T$.
	אז גם $I = \bigcap_{\alpha \in T} I_{\alpha} \triangleleft A$.
\end{proposition}
\begin{proposition}
	יהיו $I_1, \ldots, I_k \triangleleft A$ אידאלים עבור $k \in \NN$.
	אז גם $I = \sum_{i = 1}^k I_i = \{ a_1 + \cdots + a_k \mid a_i \in I_i \} \triangleleft A$.
\end{proposition}
\begin{proof}
	תרגיל.
\end{proof}
\begin{proposition}
	חיבור אידאלים נכון גם במקרה האינסופי, כלומר אם $T$ קבוצת אינדקסים ו־$I_{\alpha} \triangleleft A$ לכל $\alpha \in T$ אז גם $I = \sum_{\alpha \in T} I_{\alpha} \triangleleft A$.
	כאשר אנו מגדירים זאת עם סכימות של תת־קבוצות סופיות.
\end{proposition}
\begin{example}
	עבור $A = \ZZ$ וכן $I_i = n_i \ZZ$ מתקיים,
	\[
		\bigcap_{i = 1}^\infty I_i = \lcm\{ n_i \mid i \in T \} \ZZ,
		\qquad
		\sum_{i = 1}^\infty I_i = \gcd\{ n_i \mid i \in T \} \ZZ
	\]
\end{example}
עתה נעבור לדון במקרה שבו $A$ אידאל חילופי.
\begin{remark}
	במקרה זה כדי לבדוק ש־$\emptyset \ne I \subset A$ אידאל מספיק לבדוק ש־$I + I \subset I, A I \subset I$, כלומר השמטנו בדיקה של כפל בשני הכיוונים.
\end{remark}
\begin{definition}[אידאל נוצר]
	תהי $S \subset A$ תת־קבוצה.
	נגדיר את האידאל הנוצר על־ידי $S$ ב־$A$ על־ידי,
	\[
		(S)
		= \left\{ \sum_{i = 1}^k a_i s_i \mid a_i \in A, s_i \in S, k \in \NN \right\}
	\]
\end{definition}
\begin{proposition}
	$(S)$ הוא אכן אידאל, וכן כל אידאל של $A$ המכיל את $S$ מכיל גם את $(S)$.
\end{proposition}
\begin{proof}
	יהיו $a = \sum_{i = 1}^k a_i s_i, b = \sum_{i = 1}^l b_i s_i \in (S)$.
	אז $a + b \in (S)$ על־ידי הסכום של $(a_i + b_i) s_i$.
	בנוסף לכל $c \in A$ נקבל $c a = \sum_{i = 1}^k (c a_i) s_i \in (S)$ ולכן $(S) \triangleleft A$.

	לחלק השני יהי $S \subset I \triangleleft A$.
	לכל $a_1, \ldots, a_k \in A, s_1, \ldots, s_k \in S$ מתקיים $a_i \in A, s_i \in S \subset I$.
	לכן,
	\[
		a_i s_i \in I
		\implies \sum a_i s_i \in I
	.\]
	ולכן נובע ש־$(S) \subset I$.
\end{proof}
\begin{notation}
	נסמן $(\{a_1, \ldots, a_n\}) = (a_1, \ldots, a_n)$ ובפרט $(\{ a \}) = (a) = A a$.
\end{notation}
\begin{exercise}
	הראו כי $(a_1, \ldots, a_n) = A a_1 + \cdots + A a_n$.
\end{exercise}
\begin{definition}[אידאל ראשי]
	לכל $A$ חוג, אידאל הנוצר על־ידי איבר אחד, כלומר $I = (a)$ עבור $a \in A$, אז נאמר ש־$I$ אידאל ראשי (principal ideal).
\end{definition}
\begin{example}
	בחוג $\ZZ$ כל אידאל הוא מהצורה $(n) = n \ZZ$ ולכן ראשי.
\end{example}
\begin{proposition}

	\[
		K
		= I J
		= \{ \sum_{i = 1}^k a_i b_i \mid a_i \in I_i, b_i \in J_i, k \in \NN \}
	.\]
	אז $K \triangleleft A$.
\end{proposition}
\begin{proof}
	תרגיל.
\end{proof}
\begin{example}

\end{example}

\subsection{חוגי מנה}
בהינתן חוג $A$ ואידאל $I \triangleleft A$ נרצה להגדיר את חוג המנה $A / I$.
$A$ הוא חילופי ביחס לחיבור ולכן $I \le A$ תת־חבורה נורמלית ביחס לחיבור.
נגדיר את,
\[
	R
	= A / I
	= \{ a + I \mid a \in A \}
.\]
אם פעולת החיבור $(a + I) + (b + I) = (a + b) + I$ היא חבורה, עתה נרצה להגדיר כפל ולהפוך אותה לחוג.
\begin{notation}
	נסמן $\bar{a}$ עבור $a + I$.
\end{notation}
בהתאם מתקיים $\bar{a} = \bar{b}$ אם ורק אם $a - b \in I$.
סימון נוסף הוא $a \equiv b \mod I$.
\begin{proposition}
	עבור $\bar{a}, \bar{b} \in R$ הפעולה $\bar{a} \cdot \bar{b} = \bar{ab}$ מוגדרת היטב וכן $(R, +, \cdot)$ חוג.
\end{proposition}
\begin{proof}
	נבחר נציגים $a - a' \in I$ וכן $b - b' \in I$ ונראה כי $ab - a' b' \in I$.
	\[
		a b - a' b'
		= a \overbrace{(b - b')}^{\in I} + \overbrace{(a - a')}^{\in I} b'
		\in I
	.\]
	בהתאם באמת מצאנו שהפעולה מוגדרת היטב עבור חבורת המנה.

	החלק של בדיקת הגדרות החוג עבור $(R, +, \cdot)$ מושארת כתרגיל.
\end{proof}
\begin{definition}[חוג מנה]
	$R = A / I$ עבור $A$ חוג ו־$I \triangleleft A$ נקרא חוג המנה (quotient ring) של $A$ ב־$I$. \\
	ההעתקה $\pi : A \to A / I$ המוגרת על־ידי,
	\[
		\pi(a)
		= \bar{a}
		= a + I
		\in R
	.\]
	נקראת העתקת המנה ביחס לאידאל $I$.
\end{definition}
\begin{example}
	אם $A = \ZZ$ וכן $I = n \ZZ$ כאשר $n \in \NN_0$ אז,
	\[
		A / I
		= \ZZ / n \ZZ
	.\]
	חוג שאריות מודולו $n$.
\end{example}

\subsection{הומומורפיזמים}
\begin{definition}[הומומורפיזם חוגים]
	יהיו $A, B$ חוגים.
	פונקציה $\varphi : A \to B$ נקראת הומומורפיזם חוגים אם היא מקיימת את התכונות הבאות:
	\begin{enumerate}
		\item $\varphi : (A, +) \to (B, +)$ הומומורפיזם של חבורות, כלומר $\varphi(0_A) = 0_B$ וכן $\varphi(a \pm b) = \varphi(a) \pm \varphi(b)$ לכל $a, b \in A$
		\item $\varphi(1_A) = 1_B$
		\item פעולת הכפל משומרת, כלומר לכל $a, b \in A$ מתקיים $\varphi(a b) = \varphi(a) \cdot_B \varphi(b)$
	\end{enumerate}
\end{definition}
כאשר אנו יודעים כי הומומורפיזם הוא הומומורפיזם חוגים לעתים נשמיט את המילה חוגים.
\begin{example}
	אם $A$ חוג ו־$I \triangleleft A$ אידאל אז העתקת המנה $\pi : A \to A / I$ היא הומומורפיזם.
\end{example}
\begin{example}
	נגדיר את $\varphi : \CC \to M_2(\RR)$ על־ידי,
	\[
		\varphi(a + b i)
		= \begin{pmatrix}
			a & -b \\
			b & a
		\end{pmatrix}
	.\]
	על־ידי בדיקה פשוטה נקבל שזהו הומומורפיזם.
\end{example}
\begin{definition}[איזומורפיזם חוגים]
	הומומורפיזם חוגים נקרא איזומורפיזם חוגים אם הוא פונקציה הפיכה, ובמקרה זה גם הפונקציה ההופכית היא איזומורפיזם חוגים.
\end{definition}
\begin{notation}
	נסמן איזומורפיזם על־ידי $\varphi : A \xrightarrow{\sim} B$.
\end{notation}
\begin{definition}[חוגים איזומורפיים]
	חוגים $A, B$ יקראו איזומורפיים אם קיים $\varphi : A \xrightarrow{\sim} B$ ביניהם.
\end{definition}
\begin{definition}[גרעין]
	הגרעין של הומומורפיזם $\varphi : A \to B$ מוגדר להיות,
	\[
		\ker \varphi
		= \varphi^{-1}(\{ 0_B \})
		= \{ a \in A \mid \varphi(a) = 0 \}
	.\]
\end{definition}
\begin{proposition}
	אם $\varphi : A \to B$ הומומורפיזם אז $\ker \varphi \triangleleft A$.
\end{proposition}
\begin{proof}
	מתורת החבורות ידוע שהגרעין הוא תת־חבורה חיבורית של $A$.
	נשאר אם כן ש־$I = \ker \varphi$ סגור לכפל ב־$A$.
	יהי $a \in A$ ו־$b \in I$, אז,
	\[
		\varphi(a b)
		= \varphi(a) \varphi(b)
		= \varphi(a) \cdot 0_B
		= 0
	.\]
	ולכן $a b \in I$.
\end{proof}
\begin{definition}[תמונה]
	התמונה של הומומורפיזם $\varphi : A \to B$ היא,
	\[
		\im \varphi
		= \varphi(A)
		= \{ \varphi(a) \mid a \in A \}
	.\]
\end{definition}
\begin{definition}[תת־חוג]
	תת־חוג (sub-ring) של חוג $A$ היא תת־קבוצה $B \subset A$ כך ש־$B$ סגורה לחיבור, חיבור וכפל, וכן $1_A \in B$.
\end{definition}
\begin{exercise}
	אם $\varphi : A \to B$ הומומורפיזם של חוגים אז $\varphi(A) \subset B$ תת־חוג.
\end{exercise}
\begin{theorem}[האיזומורפיזם הראשון לחוגים]
	יהי $\varphi : A \to B$ הומומורפיזם חוגים, אז יש איזומורפיזם $A / \ker \varphi \xrightarrow{\sim} \im \varphi$ על־ידי $a + \ker \varphi \mapsto \varphi(a)$.
\end{theorem}
\begin{proof}
	מתורת החבורות נקבל שההעתקה הזו אכן מוגדרת היטב, ומהווה איזומורפיזם חבורות ביחס לפעולת החיבור.
	נשאר להראות שהיא שומרת על כפל ו־$1$.
	יהיו $a, b \in A$ אז,
	\[
		(a + \ker \varphi) (b + \ker \varphi)
		= a b + \ker \varphi
		\mapsto \varphi(a b)
		= \varphi(a) \varphi(b)
	.\]
	ולכן קיבלנו שאכן הפעולה משמרת כפל. \\
	בנוסף נבחין כי $1 + \ker \varphi$ הוא איבר היחידה של $A / \ker \varphi$ והוא נשלח ל־$\varphi(1) = 1_B$ ולכן $\varphi$ אכן איזומורפיזם חוגים.
\end{proof}
\begin{theorem}[ההתאמה לחוגים]
	יהי $A$ חוג ו־$I \triangleleft A$ אידאל, ונסמן $\pi : A \to A / I$ העתקת המנה.
	אז יש התאמה חד־חד ערכית ועל בין $\{ I \subset J \triangleleft A \}$ ל־$\{ K \triangleleft A / I \}$ המוגדרת על־ידי,
	\[
		J \mapsto \pi(J)
		= J / I
	.\]
\end{theorem}
לפני שניגש להוכחת המשפט נוכיח טענת עזר חשובה.
\begin{lemma}
	יהי $\varphi : A \to B$ הומומורפיזם חוגים, אז,
	\begin{enumerate}
		\item אם $I \triangleleft B$ אז $\varphi^{-1}(I) \triangleleft A$
		\item אם $I \triangleleft A$ וכן $\varphi$ על אז $\im \varphi \triangleleft B$
	\end{enumerate}
\end{lemma}
\begin{proof}
	החלק הראשון מושאר כתרגיל.

	אנו יודעים כי $\varphi(I)$ תת־חבורה חיבורית של $B$, עלינו להראות שגם $B \varphi(I), \varphi(I) B \subset \varphi(I)$.
	מתקיים,
	\[
		B \varphi(I)
		= \varphi(A) \varphi(I)
		= \varphi(A I)
		= \varphi(I)
	.\]
	כאשר המעבר הראשון נובע מהעובדה ש־$\varphi$ על, והכיוון השני זהה.
\end{proof}
נחזור למשפט ההתאמה.
\begin{proof}[הוכחת המשפט]
	ידוע מתורת החבורות שההתאמה שתיארנו קיימת בין תת־חבורות חיבוריות $\{ I \subset H \subset A \}$ לבין תת־חבורות $\{ Q \subset A / I \}$.
	נותר להראות שהתאמה זו משמרת אידאלים.
	$\pi$ היא על ולכן מטענת העזר נובע שאם $I \subset J$ אידאל אז $\pi(J)$ וכן אם $K \triangleleft A / I$ אז $\pi^{-1}(K) \triangleleft A$ כנדרש.
\end{proof}
\begin{remark}
	$\pi(J) = J / I$ הוא אידאל ב־$A / I$.
\end{remark}
\begin{theorem}[האיזומורפיזם השלישי]
	יהי $A$ חוג ויהי $I \triangleleft A$ אידאל, וכן $I \subset J \triangleleft A$ אידאל נוסף.
	אז מתקיים,
	\[
		A / J
		\cong (A / I) / (J / I)
	.\]
	על־ידי המיפוי,
	\[
		a + J
		\mapsto \pi(a) + J / I
	.\]
\end{theorem}
\begin{proof}
	מתורת החבורות קיים איזומורפיזם כזה, וקל לבדוק שוב את התנאים הנוספים הנדרשים.
\end{proof}

\begin{theorem}[האיזומורפיזם השני]
	יהי $A$ חוג, $B \subset A$ תת־חוג ו־$I \triangleleft A$ אידאל.
	\begin{enumerate}
		\item $I + B$ הוא תת־חוג של $A$
		\item $B \cap I \triangleleft B$ וכן $I \triangleleft I + B$
		\item $B / (I \cap B) \cong (I + B) / I$ נתון על־ידי $b + (I \cap B) \mapsto b + I$
	\end{enumerate}
\end{theorem}
\begin{proof}
	\textbf{1.}
	יהיו $a, b \in I + B$, אז ברור ש־$a \pm b \in I + B$.
	נסמן $a = c + d, b = e + f$ עבור $c, e \in I$ ו־$d, f \in B$, אז מתקיים,
	\[
		a b
		= (c + d)(e + f)
		= \overbrace{c e + d e + cf}^{\in I} + \overbrace{df}^{\in B}
		\in I + B
	.\]
	וכן $1 = 1 + 0 \in I + B$ ולכן $I + B$ אכן תת־חוג.

	\textbf{2.}
	ניתן לבדיקה ישירה.

	\textbf{3.}
	מתורת החבורות ידוע שההעתקה הנתונה מוגדרת היטב ומהווה איזומורפיזם חבורות, ולכן מספיק לבדוק שהיא שומרת על כפל ויחידה (תרגיל).
\end{proof}

\subsection{אידאלים ראשוניים}
באנגלית נקרא prime ideal, בחלק זה נניח ש־$A$ הוא חוג \textbf{חילופי}.

\begin{definition}[אידאל ראשוני]
	$P \triangleleft A$ נקרא אידאל \textbf{ראשוני} אם:
	\begin{enumerate}
		\item $P \ne A$, כלומר זהו אידאל ממש
		\item לכל $a, b \in A$ אם $a b \in P$ אז או ש־$a \in P$ או ש־$b \in P$, כלומר המצב ההפוך לסגירות לכפל של האידאל
	\end{enumerate}
\end{definition}
\begin{proposition}[תנאי שקול לאידאל ראשוני]
	$P$ הוא ראשוני אם ורק אם $A / P$ תחום.
\end{proposition}
\begin{proof}
	נניח ש־$A / P$ תחום, לכן $A / P \ne 0$, או באופן שקול $A \ne P$.
	בנוסף אם $a, b \in A$ כך ש־$a b \in P$ אז $\pi(a b) = 0$ עבור $\pi : A \to A / P$ העתקת המנה, ו־$A / P$ תחום ולכן $\pi(a) = 0$ או $\pi(b) = 0$,
	כלומר $a \in P$ או $b \in P$ ולכן $P$ ראשוני לפי ההגדרה.

	הכיוון השני זהה.
\end{proof}
\begin{example}
	נגדיר $A = \ZZ$ חוג השלמים, ונבדוק עבור אילו טבעיים $n \in \NN_0$ האידאל $(n) = n \ZZ$ הוא ראשוני. \\
	אם $n = 0$ אז $\ZZ / (0) \cong \ZZ$ וזהו תחום ולכן $(0)$ ראשוני. \\
	אם $n = 1$ אז $\ZZ / (1) \cong \{ 0 \}$ וזהו אינו תחום ולכן $(1)$ לא ראשוני. \\
	נניח עתה ש־$n$ מספר ראשוני, אז $\ZZ / (p) \cong \FF_p$ הוא שדה ובפרט תחום ולכן $(n)$ ראשוני. \\
	אם $n$ מספר פריק גדול וגדול מ־$1$, אז קיימים $a, b > 1$ עבורם $n = a b$ אז גם $a b \in (n)$ אבל $a, b \notin (n)$ ולכן $(n)$ לא ראשוני.

	מצאנו שב־$\ZZ$ האידאלים הראשוניים הם אידאל האפס ו־$(p)$ לכל $p$ ראשוני.
\end{example}
\begin{definition}[אידאל מקסימלי]
	יהי $A$ חוג כלשהו (לא בהכרח חילופי), $M \triangleleft A$ נקרא אידאל \textbf{מקסימלי} אם הוא מתקיים:
	\begin{enumerate}
		\item $M \ne A$
		\item לא קיים $A \ne I \triangleleft A$ כך ש־$M \subsetneq I$, או באופן שקול כל אידאל $M \subset I \triangleleft A$ מקיים $I = A$ או $I = M$
	\end{enumerate}
\end{definition}
\begin{lemma}
	יהי $A \ne 0$ חוג חילופי, אז $A$ שדה אם ורק אם כל האידאלים ב־$A$ הם $(0), (1)$.
\end{lemma}
\begin{proof}
	יהי $A$ שדה ויהי $I \triangleleft A$.
	אם $I = (0)$ אז סיימנו ולכן נניח ש־$I \ne 0$ ולכן קיים $a \in I$, אבל אז גם $1 = a \cdot a^{-1} \in I$ ולכן $I = (1)$.

	בכיוון ההפוך נניח שכלל האידאלים הם $(0), (1)$ וכן ש־$0 \ne 1$ ונראה שכל $a \ne 0$ הפיך.
	$(a) \ne (0)$ ולכן $(a) = (1)$, כלומר $1 = a b$ ו־$a$ בהתאם הפיך.
\end{proof}
\begin{proposition}
	$M \triangleleft A$ מקסימלי אם ורק אם $A / M$ שדה.
\end{proposition}
\begin{proof}
	$M$ מקסימלי אם ורק אם כל האידאלים בין $M$ ו־$A$ הם בדיוק $M, A$ ובנוסף $M \ne A$, ממשפט ההתאמה טענה זו שקולה לכך שכל האידאלים ב־$A / M$ הם $(0), (1)$ והם שונים, וזוהי בדיוק הלמה אותה הוכחנו זה עתה.
\end{proof}
\begin{corollary}
	כל אידאל מקסימלי הוא ראשוני.
\end{corollary}
\begin{proof}
	אם $M$ מקסימלי אז $A / M$ שדה ולכן בפרט $A / M$ הוא תחום ולכן $M$ ראשוני.
\end{proof}
נראה שטענה זו נכונה רק בכיוון הזה.
\begin{example}
	נבחן שוב את $A = \ZZ$ ונבחין כי $(0)$ ראשוני אבל לא מקסימלי כי $(0) \subsetneq (2) \ne (1)$. \\
	אם $p$ מספר ראשוני אז נראה ש־$(p)$ מקסימלי.
	אם $(p) \subsetneq (n) \subsetneq (1)$ אז $n \mid p$ וכן $n > 1$ וזו סתירה לראשיות של $(p)$.
	נסיק שכל האידאלים ב־$\ZZ$ הם מהצורה $(p)$ עבור $p$ מספר ראשוני.
	נובע גם ש־$\ZZ / (p)$ הוא אכן שדה, וש־$\FF_p$ שדה אם ורק אם $p$ ראשוני.
\end{example}
\begin{theorem}[האידאל המקסימלי]
	בכל $A \ne 0$ קיים אידאל מקסימלי.
	נקרא גם משפט Krull.
\end{theorem}

\subsection{אידאלים מקסימליים}
נוכיח את משפט האידאל המקסימלי.
\begin{proof}
	נסמן $S = \{ I \triangleleft A \mid I \ne A \}$ כל האידאלים ב־$A$ ונבדוק את תנאי הלמה של צורן.
	תהי $T \subset S$ שרשרת ביחס ההכלה וניקח את $I = \{ 0 \} \cup \bigcup T$ ונראה ש־$I \in S$ וכן שהוא חסם עליון של $T$.
	אכן $J \subset I$ לכל $J \in T$, ונראה ש־$I \triangleleft A, I \ne A$. \\
	יהיו $a, b \in I$, ונניח בלי הגבלת הכלליות ש־$T$ לא ריקה. קיימים $J, K \in S$ כך ש־$b \in K, a \in J$.
	$T$ שרשרת ולכן $J \subset K$ או $K \subset J$ ונניח בלי הגבלת הכלליות ש־$K \subset J$.
	אז $a, b \in J$ ולכן $a + b \in J \subset I$ כנדרש.
	אם $c \in I$ ו־$a \in A$ אז קיים $J \in T$ כך ש־$c \in J$ ולכן גם $a c \in J \subset I$ ולכן נוכל להסיק ש־$I \triangleleft A$.
	בנוסף $1 \notin J$ לכל $J \in T$ ולכן גם $1 \notin J$ ובפרט $J \ne A$.
	בהתאם $J \in S$ הוא אכן חסם עליון ולכן תנאי הלמה של צורן מתקיימים וקיים אידאל מקסימלי.
\end{proof}
\begin{corollary}
	יהי $A$ חוג ויהי $I \triangleleft A$ אידאל ממש, אז קיים אידאל $I \subset M \triangleleft A$ מקסימלי.
\end{corollary}
\begin{proof}
	לפי משפט ההתאמה אידאלים מקסימליים $I \subset M \triangleleft A$ מתאימים לאידאלים מקסימליים ב־$A / I$ ולפי משפט האידאל המקסימלי קיים $N \triangleleft A / I$ מקסימלי.
\end{proof}

\subsection{חוגי פולינומים}
לפרק זה נניח כי כל החוגים הם חילופיים.
נבחין כי כמעט לכל מתייחס כאן לתמיד למעט מספר סופי של מקרים. \\
בנוסף אם $(A, +)$ חבורה חילופית ו־$f : X \to A$ פונקציה עבור קבוצה כלשהי $X$, כך ש־$f(x) = 0$ כמעט לכל $x$ אז נסמן,
\[
	\sum_{x \in X} f(x)
	= \sum_{x \in X \setminus f^{-1}(\{ 0 \})} f(x)
\]
\begin{definition}[מונום]
	תהי קבוצה $X$, נחשוב עליה כאוסף של משתנים בלתי־תלויים.
	מונום (monomial) מעל $X$ הוא פונקציה $m : X \to \NN_0$ כך ש־$m(x) = 0$ כמעט לכל $x$.
\end{definition}
נסמן $m = x^2 y^3 z$ עבור המונום כך ש־$f(x) = 2, f(y) = 3, f(z) = 1$ ולכל משתנה אחר $f(s) = 0$.
נסמן ב־$1$ את המונום $m \equiv 0$.
\begin{notation}
	נסמן ב־$\Mon(X)$ את אוסף כל המונומים במשתנים $X$.
\end{notation}
\begin{definition}[כפל מונומים]
	אם $m_1, m_2 \in \Mon(X)$ אז נגדיר $m_1 \cdot m_2$ כמונום $m(x) = m_1(x) + m_2(x)$ לכל $x \in X$.
\end{definition}
\begin{example}
	בסימון שלנו $x y^2 \cdot x^2 y^3 = x^3 y^5$.
\end{example}
\begin{definition}[פולינום]
	נגדיר פולינום (polynomial) עם במשתנים $X$ בחוג (חילופי) $A$ כפונקציה $p : M \to A$ שמתאפסת לכמעט כל מונום.
\end{definition}
\begin{notation}
	נסמן פולינומים כסכום של מונומים עם מקדמים מ־$A$, כלומר לדוגמה $a x^2 + b x y z + c$ עבור $a, b, c \in A$ הוא הפולינום המוגדר על־ידי,
	\[
		p(x^2) = a,
		\quad
		p(x y z) = b,
		\quad
		p(1) = c,
		\quad
		p(s) = 0
	\]
\end{notation}
\begin{definition}[קבוצת הפולינומים]
	נסמן את קבוצת כל הפולינומים במשתנים $X$ מעל $A$ על־ידי $A[X]$. \\
	נגדיר חיבור פולינומים על־ידי $(p_1 + p_2)(m) = p_1(m) + p_2(m)$ וכפל על־ידי,
	\[
		(p_1 \cdot p_2)(m)
		= \sum_{\substack{m_1, m_2 \in \Mon(X) \\ m_1 m_2 = m}} p_1(m_1) p_2(m_2)
	\]
	ולבסוף נגדיר את פולינום ה־$1$ כפולינום $p(m) = \indicator_{\{ m = 1 \}}$.
\end{definition}
\begin{proposition}
	$(A[X], +, \cdot)$ הוא חוג חילופי כאשר פולינומי ה־$0$ וה־$1$ הם בדיוק $0_{A[X]}$ ו־$1_{A[X]}$.
\end{proposition}
\begin{proof}
	תרגיל.
\end{proof}
\begin{notation}
	$A[\{ x_1, \ldots, x_n \}] = A[x_1, \ldots, x_n]$.
\end{notation}
\begin{proposition}
	$A[x_1, \ldots, x_n] \cong A[x_1, \ldots, x_{n - 1}][x_n]$.
\end{proposition}
גם טענה זו מושארת כתרגיל.

\subsection{אלגברות}
\begin{definition}[אלגברה]
	יהי $A$ חוג חילופי, אלגברה (algebra) חילופית מעל $A$ היא זוג $(B, \varphi)$ כאשר $B$ חוג חילופי ו־$\varphi : A \to B$ הומומורפיזם.
\end{definition}
אפשר לחשוב על $B$ כחוג חילופי עם פעולה נוספת של כפל בסקלר מ־$A$, שכן,
\[
	b \in B, \lambda \in A
	\implies \lambda b := \varphi(\lambda) b
\]
\begin{example}
	כל חוג חילופי $A$ הוא אלגברה מעל $\ZZ$ על־ידי ההומומורפיזם $n \mapsto n \cdot 1_A$.
\end{example}
\begin{example}
	אם $A$ חוג ו־$I$ אידאל אז $(A / I, \pi)$ עבור $\pi$ העתקת המנה, הוא אלגברה מעל $A$.
\end{example}
\begin{example}
	אם $A$ חוג ו־$X$ אוסף משתנים אז $A[X]$ אלגברה יחד עם ההומומורפיזם $a \mapsto a \cdot 1$ עבור $1$ מונום היחידה, בדרך־כלל נשמיט את ה־$1$.
\end{example}
\begin{definition}[הצבה בפולינום]
	יהי $A$ חוג (חילופי) ו־$(B, \varphi)$ אלגברה מעל $A$, $X$ קבוצת משתנים ונניח ש־$X$ סופית (אך ההכללה פשוטה) ו־$X = x_1, \ldots, x_n$.
	\textbf{הצבה} של ערכים מ־$B$ במשתנים $X$ היא פונקציה $f : X \to B$ ונסמן $f(x_i) = b_i$.

	\textbf{הצבה במונום} $m \in \Mon(X)$ מוגדרת על־ידי,
	\[
		m(b_1, \ldots, b_n)
		= \prod_{i = 1}^n b_i^{m(x_i)} \in B
	\]

	\textbf{הצבה בפולינום} $p$ תוגדר על־ידי,
	\[
		p(b_1, \ldots, b_n)
		= \sum_{m \in \Mon(X)} \varphi(p(m)) m(b_1, \ldots, b_n) \in B
	\]
\end{definition}
\begin{example}
	אם $p = x_1^2 + a x_2^2 + 1$ אז $p(b_1, b_2) = b_1^2 + a b_2^2 + 1$.
\end{example}
\begin{theorem}[התכונה האוניברסלית של חוג הפולינומים]
	יהי $A$ חוג ו־$X = \{x_1, \ldots, x_n\}$, תהי $(B, \varphi)$ אלגברה מעל $A$, ו־$b_1, \ldots, b_n \in B$.
	אז מתקיים,
	\begin{enumerate}
		\item פעולת ההצבה $\psi(p) = p(b_1, \ldots, b_n)$ היא הומומורפיזם $\psi : A[X] \to B$ כך שמתקיים $\psi(a) = \varphi(a)$ לכל $a \in A$ ובנוסף $\psi(x_i) = b_i$
		\item $\psi$ הוא ההומומורפיזם היחיד $A[X] \to B$ כך ש־$\psi |_A = \varphi$ ו־$\psi(x_i) = b_i$ לכל $i$
	\end{enumerate}
\end{theorem}
\begin{proof}
	תחילה עלינו להראות ש־$\psi$ הומומורפיזם.
	תחילה מראים שהוא שומר על כפל מונומים ואז מסיקים שהוא שומר על כפל פולינומים (תרגיל).
	$\psi(x_i) = b_i$: אם $a$ קבוע אז $\psi(a) = \varphi(a) \cdot 1 = \varphi(n)$.

	נעבור ליחידות:
	נובע מכך שכל פולינום הוא סכום של מונומים כפול מקדמים וכל מונום הוא מכפלה של המשתנים.
\end{proof}

\subsection{נילפוטנטים ורדיקלים}
\begin{definition}[נילפוטנט]
	יהי $A$ חוג.
	איבר $a \in A$ יקרא נילפוטנט (nilpotent) אם קיים $n \in \NN$ כך ש־$a^n = 0$.
\end{definition}
\begin{definition}[רדיקל]
	$\Nil(A) = \{ a \in A \mid a \text{ is nilpotent} \}$ נקרא הרדיקל של $A$.
\end{definition}
\begin{example}
	נחשב את הרדיקל של הבאים:
	\begin{enumerate}
		\item $a = 0$ תמיד נילפוטנט
		\item אם $A = \ZZ / (n)$ עבור $n \in \NN$ אז $m \mod n$ נילפוטנט אם ורק אם $m$ מתחלק בכל גורם ראשוני של $n$
		\item אם $A = \ZZ[x] / (x^n)$ אז $a = (x \mod (x^n))$ הוא נילפוטנט כי $a^n = 0$
	\end{enumerate}
\end{example}
\begin{remark}
	אם $A \ne 0$ אז כל נילפוטנט הוא מחלק 0.
	אם $a = 0$ הוא מחלק 0, אחרת יהי $n \in \NN$ מינימלי כך ש־$a^n = 0$, ואז $a \cdot a^{n - 1} = 0$.
\end{remark}
\begin{theorem}[שקילות רדיקל וחיתוך האידאלים הראשוניים]
	נניח ש־$A \ne 0$, ונסמן $\Pp$ קבוצת כל האידאלים הראשוניים ב־$A$, אז,
	\[
		\Nil(A)
		= \bigcap \Pp
	\]
	בפרט $\Nil(A)$ הוא אידאל.
\end{theorem}
\begin{proof}
	תחילה נראה ש־$\Nil(A) \subset \bigcap \Pp$, כלומר שלכל $P \in \Pp$ ו־$a$ נילפוטנט מתקיים $a \in P$.
	תהי $\pi : A \to A / P$ העתקת המנה, אם $a^n = 0$ אז ${(\pi(a))}^n = \pi(a^n) = \pi(0) = 0$ ולכן גם $\pi(a)$ נילפוטנט ב־$A / P$, בפרט הוא מחלק 0.
	אבל $A / P$ תחום ולכן $\pi(a) = 0$ בלבד.

	נראה את הכיוון ההפוך, כלומר שאם $a$ איננו נילפוטנט אז קיים $P \triangleleft A$ ראשוני שלא מכיל את $a$.
	נגדיר את קבוצת האידאלים,
	\[
		S
		= \{ I \triangleleft A \mid I \cap \{1, a, a^2, \ldots\} = \emptyset \}
	\]
	אז $(S, \subset)$ סדר חלקי, ונבדוק שהיא מקיימת את תנאי הלמה של צורן.
	אם $T \subset S$ שרשרת אז נבחר $I = \bigcup I \cup \{ 0 \}$, זהו אידאל כאיחוד אידאלים, וכן הוא חסם עליון של השרשרת כאיחודה.
	אם כך נובע מהלמה של צורן שקיים $P \in S$ מקסימלי.
	אז $P \triangleleft A$, $P \cap \{1, a, a^2, \ldots\} = \emptyset$ וכן שאם $P \subsetneq Q \triangleleft A$ אז בהכרח $Q$ מכיל חזקה של $a$, ונראה ש־$P$ ראשוני.
	יהיו $b, c \notin A \setminus P$ ונראה ש־$b c \notin P$.
	נניח בשלילה ש־$b c \in P$ ונבחן את $Q = P + (b), Q' = P + (c)$, אז $P \subsetneq Q, Q'$ ולכן מכילים חזקות של $a$, כלומר קיימים $a^n \in P + (b), a^m \in P + (c)$ ל־$n, m \in \NN$.
	אם $p_1, p_2 \in P, \lambda_1, \lambda_2 \in A$ כך ש־$a^n = p_1 + \lambda_1 b, a^m = p_2 + \lambda_2 c$ אז,
	\[
		a^{m + n}
		= (p_1 + \lambda_1 b)(p_2 + \lambda_2 c)
		= p_1 p_2 + \lambda_1 b p_2 + \lambda_2 c p_1 + \lambda_1 \lambda_2 b
		\in P
	\]
	וקיבלנו ש־$a^{m + n} \in P$ וזו סתירה.
\end{proof}
\begin{definition}[רדיקל של אידאל]
	הרדיקל של אידאל $I \triangleleft A$ מוגדר להיות,
	\[
		\sqrt{I}
		= \{ a \in A \mid \exists n \in \NN,\ a^n \in I \}
		\subset A
	\]
\end{definition}
\begin{remark}
	$\sqrt{(1)} = (1)$ וכן הרדיקל של $A$ הוא בדיוק $\sqrt{(0)}$.
\end{remark}
\begin{proposition}
	אם $I \ne (1)$ אז מתקיים $\sqrt{I} = \bigcap \Pp$ עבור $\Pp$ קבוצת האידאלים $I \subset P \triangleleft A$ הראשוניים.
\end{proposition}
\begin{proof}
	תהי $\pi : A \to A / I$ העתקת המנה,
	\[
		\sqrt{I}
		= \pi^{-1}(\Nil(A / I))
	\]
	ולכן,
	\[
		\sqrt{I}
		= \pi^{-1}(\bigcap_{\substack{Q \triangleleft A / I \\ \text{prime}}} Q)
		= \bigcap_{\substack{Q \triangleleft A / I \\ \text{prime}}} \pi^{-1}(Q)
		= \bigcap \Pp
	\]
	כאשר הטענה נובעת ממשפט ההתאמה ושימוש באידאלים ראשוניים מתאימים לראשוניים ממשפט האיזומורפיזם השלישי.
\end{proof}
\begin{definition}[רדיקל ג'ייקובסון]
	רדיקל ג'ייקובסון של חוג $A \ne 0$ מוגדר להיות,
	\[
		\Jj(A)
		= \bigcap_{\substack{M \triangleleft A \\ \text{maximal}}} M
		\triangleleft A
	\]
	אם $A = 0$ אז נגדיר $\Jj = (0)$. \\
	אם $I \triangleleft A$ אז נגדיר את רדיקל ג'ייקובסון של $I \ne (1)$ להיות,
	\[
		\Jj(I)
		= \bigcap_{\substack{I \subset M \triangleleft A \\ \text{maximal}}} M
		\triangleleft A
	\]
	ואם $I = (1)$ נגדיר את $\Jj(I) = (1)$.
\end{definition}
\begin{remark}
	$\Nil(A) \subset \Jj(A)$ שכן כל אידאל מקסימלי הוא ראשוני.
\end{remark}
לפי משפט ההתאמה $\Jj(I) = \pi^{-1}(\Jj(A / I))$.
\begin{proposition}[אפיון איברי רדיקל ג'ייקובסון]
	אם $A$ חוג אז,
	\[
		\Jj(A)
		= \{ a \in A \mid \forall b \in A,\ 1 - a b \in A^\times \}
	\]
\end{proposition}
\begin{proof}
	יהי $a \in A$ כך שלכל $b \in A$, $1 - a b$ הפיך, ונראה ש־$1 - a b \in \Jj(A)$, כלומר נראה ש־$1 - a b \in M$ לכל $M \triangleleft A$ מקסימלי.
	יהי $M \triangleleft A$ מקסימלי, אם $a \notin M$ אז $M \subsetneq M + (a)$ ולכן $M + (a) = (1)$, כלומר $1 \in M + (a)$ ונרשום $1 = m + b a$ כאשר $m \in M$ ו־$b \in A$.
	אז $1 - b a = m \in M$ ואינו הפיך בסתירה (אחרת $M = A$).

	בכיוון ההפוך יהי $a \in \Jj(A)$ ו־$b \in A$ ונראה ש־$1 - a b$ הפיך.
	אם הוא אינו הפיך אז $(1 - a b) \ne (1)$ ולכן קיים אידאל מקסימלי $(1 - a b) \subset M$ אז $a \in \Jj(A) \subset M$, כלומר $a, 1 - a b \in M$ ולכן $1 = b a + 1 - a b \in M$ בסתירה.
\end{proof}

\subsection{מכפלת חוגים}
בפרק זה לא נניח שכל חוג הוא חילופי.
יהיו $A, B$ חוגים, נגדיר מבנה של חוג על $A \times B$ על־ידי,
\[
	(a_1, b_1) + (a_2, b_2)
	= (a_1 + b_1, a_2 + b_2),
	\qquad
	(a_1, b_1) \cdot (a_2, b_2)
	= (a_1 \cdot b_1, a_2 \cdot b_2)
\]
קל לבדוק ש־$A \times B$ חוג עם פעולות אלה והקבועים $0 = (0_A, 0_B), 1 = (0_A, 0_B)$. \\
באופן דומה ניתן להגדיר מכפלה של אוסף כלשהו ${\{ A_{\alpha} \}}_{\alpha \in I}$ של חוגים,
\[
	\prod_{\alpha \in I} A_{\alpha}
	= \left\{ f : I \to \bigcup_{\alpha \in I} A_{\alpha} \mid \forall \alpha \in I,\ f(\alpha) \in A_{\alpha} \right\}
\]
והפעולות,
\[
	(f + g)(\alpha)
	= f(\alpha) + g(\alpha),
	\quad
	(f g)(\alpha) = f(\alpha) \cdot g(\alpha)
\]
$A_{\alpha}$ חילופיים אם ורק אם $\prod_{\alpha \in I} A_{\alpha}$ חילופי.

יהי $A$ חוג חילופי.
\begin{definition}[אידאלים זרים]
	אידאלים $I_1, \ldots, I_k \triangleleft A$ נקראים זרים (prime) אם $I_1 + \cdots + I_k = (1)$ והם זרים בזוגות, כלומר אם $I_i + I_j = (1)$ לכל $i \ne j$.
\end{definition}
\begin{notation}
	נכתוב $(I, J)$ במקום $I + J$ ונכתוב $I = 1$ במקום $I = (1)$. \\
	בעזרת סימונים אלה אידאלים $I_1, \ldots, I_k$ זרים אם ורק אם $(I_1, \ldots, I_k) = 1$.
\end{notation}
\begin{example}
	$A = \ZZ$ ו־$I_i = (n_i)$ עבור $n_i \in \NN_0$ אז,
	\[
		(I_1, \ldots, I_k)
		= (\gcd(n_1, \ldots, n_k))
	\]
	ובפרט הם זרים אם ורק אם $n_1, \ldots, n_k$ זרים.
\end{example}
\begin{proposition}
	יהיו $I, J, K \triangleleft A$.
	\begin{enumerate}
		\item אם $(I, J) = 1$ אז $I \cap J = I J$
		\item אם $(I, K) = (I, J) = 1$ אז $(I, J K) = 1$
	\end{enumerate}
\end{proposition}
\begin{proof}
	\textbf{1.}
	לכל זוג אידאלים $I J \subset I \cap J$. נניח ש־$I + J = (1)$, כלומר $1 = a + b$ עבור $a \in I, b \in J$.
	אם $c \in I \cap J$ אז נציג $c = a c + b c$, אבל $a c \in I J$ וכן $b c \in I J$ ולכן גם $c \in I J$, כלומר $I \cap J \subset I J$.

	\textbf{2.}
	נניח ש־$I + J = I + K = 1$ ושוב נסמן איברים $a, b \in I, c \in J, d \in K$ כך ש־$a + c = 1 = b + d$.
	\[
		1
		= (a + c)(b + d)
		= \overbrace{a b + c b + a d}^{\in I} + \overset{\in J K}{c d}
		\in I + J K
	\]
	ולכן $I + J K = (1)$.
\end{proof}
\begin{corollary}
	אם $I_1, \ldots, I_k$ אידאלים זרים בזוגות אז $I_1 \cap \cdots \cap I_k = I_1 \cdots I_k$ וכן $(I_1, I_2 \cdots I_k) = 1$.
\end{corollary}
\begin{proof}
	הטענה האחרונה ואינדוקציה.
\end{proof}
\begin{theorem}[השאריות הסיני]
	יהי $A$ חוג חילופי ויהיו $I_1, \ldots, I_k \triangleleft A$ זרים בזוגות.
	אז קיים איזומורפיזם,
	\[
		A / I_1 \cdots I_k
		\cong (A / I_1) \times \cdots (A / I_k)
	\]
	על־ידי המיפוי,
	\[
		a \mod I_1 \cdots I_k
		\mapsto (a \mod I_1, \ldots, a \mod I_k)
	\]
\end{theorem}
\begin{proof}
	נגדיר $\varphi : A \to \bigcap_{i = 1}^k A / I_i$ על־ידי $\varphi(a) = (a \mod I_1, \ldots, a \mod I_k)$. \\
	זהו הומומורפיזם עם גרעין $\ker \varphi = \{ a \in A \mid \forall i,\ a \in I_i \}$.
	ממשפט האיזומורפיזם הראשון לחוגים נקבל $A / \ker \varphi \cong \im \varphi$ ונשאר להראות ש־$\im \varphi \prod_{i = 1}^k A / I_i$, כלומר ש־$\varphi$ היא על. \\
	נוכיח זאת באינדוקציה על $k$, המקרה $k = 1$ ברור, ולכן נניח $k > 1$ ושהטענה נכונה עבור $k - 1$.
	יהיו $a_1, \ldots, a_k \in A$, לפי הנחת האינדוקציה קיים $b \in A$ כך ש־$b \equiv a_i \mod I_i$ לכל $1 \le i < k$.
	מהמסקנה הקודמת אנחנו יודעים ש־$(I_1 \cdots I_{k - 1}, I_k) = 1$, כלומר קיימים $u \in I_1 \cdots I_{k - 1}, v \in I_k$ כך ש־$u + v = 1$.
	אז $u \equiv 0, v \equiv 1 \mod I_1 \cdots I_{k - 1}$ וכן $u \equiv 1, v \equiv 0 \mod I_k$.
	נגדיר את הצירוף $a = v b + u a_k$, אז $a \equiv b \mod I_1 \cdots I_k$, ובאופן דומה גם,
	\[
		a
		= v b + u a_k
		\equiv 0 \cdot b + 1 \cdot a_k
		\equiv a_k \mod I_k
	\]
	כלומר $a \equiv b \equiv a_i \mod I_i$. קיבלנו ש־$a \equiv a \mod I_i$ לכל $1 \le i \le k$, כלומר $\varphi$ היא אכן על.
\end{proof}

\end{proposition}
\begin{proof}
	בכיוון הראשון, יהי $P \triangleleft K[\bar{x}]$ אידאל ראשוני, ונסמן $V = Z(P)$, נניח בשלילה ש־$V = U \cup W$ עבור קבוצות אלגבריות $U, W \subsetneq V$.
	$P \subsetneq I(U), I(W)$ (שכן ההתאמה הופכת הכלה) ונבחר $f \in I(U) \setminus P, g \in I(W) \setminus P$, ונבחין כי $f$ מתאפס על $U$ ו־$g$ מתאפס על $W$ ולכן $f g$ מתאפס על $V$.
	לכן $f g \in I(V) = I(Z(P)) = P$ בסתירה לראשוניות $P$.

	בכיוון השני נניח ש־$P \triangleleft K[\bar{x}]$ רדיקלי אבל אינו ראשוני, ונסמן $V = Z(P)$.
	יהיו $f, g \in K[\bar{x}]$ כך ש־$f, g \in P$ אבל $f g \in P$.
	נגדיר,
	\[
		U = Z(P \cup \{ f \}),
		\qquad
		W = Z(P \cup \{ g \})
	\]
	ונראה כי $V = U \cup W$ ו־$U, W \subsetneq V$, כלומר $V$ פריקה.
	$f$ מתאפסת על $U$ ולא על $V$ ולכן $U \subsetneq V$, אותו הדבר נכון גם ל־$W$, כלומר $W \subsetneq V$.
	אם $a \in V$ אז כל $h \in P$ מתאפס על $a$, וכן $f g \in P$ ולכן $(f g)(a) = 0$, כלומר $f(a) = 0$ או $g(a) = 0$.
	במקהה הראשון $a \in U$ ובמקרה השני $a \in W$, כלומר $V = U \cap W$ פריקה.
\end{proof}

\subsection{שיכון שדות בשדות סגורים אלגברית}
המטרה שלנו היא להוכיח את המשפט,
\begin{theorem}[קיום סגור אלגברי]
	אם $K$ שדה אז קיים שדה $L$ ושיכון $K \hookrightarrow L$ כהומומורפיזם שדות.
\end{theorem}
כדי לעשות זאת ננסח ונוכיח מספר למות.
\begin{lemma}
	יהי $K$ שדה ו־$A \ne 0$ חוג, אם $\varphi : K \to A$ הומומורפיזם, אז הוא חד־חד ערכי.
\end{lemma}
\begin{proof}
	נבחן את $\ker \varphi$, זהו אידאל של $K$, אך ב־$K$ כל האידאלים הם $(0)$ או $(1)$.
	אם $\ker \varphi = (1)$ אז בפרט $1 \in \ker \varphi$  אבל $\varphi(1) = 1_A \ne 0_A$ וזו סתירה, לכן $\ker \varphi = (0)$, כלומר $\varphi$ חד־חד ערכי.
\end{proof}
\begin{lemma}
	יהיו $f_1, \ldots, f_n \in K[x] \setminus K$ פולינומים, אז קיים שיכון $K \hookrightarrow L$ עבור $K$ שדה כך שב־$K$ לכל $f_i$ יש שורש.
\end{lemma}
\begin{proof}
	מספיק לטפל במקרה של $n = 1$, שאר המקרים יתקבלו באינדוקציה.
	נגדיר את $A = K[x] / (f_1)$, אז בחוג זה מתקבל ש־$x$ הוא שורש.
	$(f_1) \ne (1)$ שכן $f_1$ לא קבוע, לכן $A \ne 0$.
	נבחר $M \triangleleft A$ מקסימלי, לכן $A / M$ הוא שדה,
	\[
		K
		\hookrightarrow K[x]
		\hookrightarrow K[x] / (f_1)
		= A
		\to A / M
		= L
	\]
	התמונה של $x \mod (f_1)$ היא השורש של $f_1$ ב־$L$, לפי הלמה הראשונה הרכבת הומומורפיזמים כזאת היא חד־חד ערכית, כלומר $K \hookrightarrow L$.
\end{proof}
\begin{proposition}
	יהי $K$ שדה, אז קיים שדה $L$ עם שיכון $K \hookrightarrow L$ כך שלכל $f \in K[x] \setminus K$ יש שורש ב־$L$.
\end{proposition}
נראה את ההוכחה של Emil Artin.
\begin{proof}
	יהי $\bar{x} = \{ x_f \mid f \in K[x] \setminus K \}$ אוסף משתנים.
	נבחן את $A = K[\bar{x}] / (f(x_k) \mid f \in K[x] \setminus K)$, לכן $x_f \mod I$ הוא שורש של $f$ ב־$A$.
	תחילה נראה ש־$A \ne 0 \iff I = (1)$, נניח בשלילה ש־$I = (1)$, כלומר,
	\[
		1
		= \sum_{i = 1}^n h_i(\bar{x}) f_i(x_{f_i})
	\]
	קיים שיכון $K \hookrightarrow F$ עבור $F$ שדה שבו ל־$f_1, \ldots, f_n$ יש שורש (וזוהי בדיוק הלמה הקודמת).
	$\alpha_i \in F$ כך ש־$f(\alpha_i) = 0$ ונציב $x_{f_i} \mapsto \alpha_i$ וכן $x_g \mapsto 0$ אם $g \in \{ f_1, \ldots, f_n \}$, לכן,
	\[
		1
		= \sum_{i = 1}^n h_i(\alpha_1, \ldots, \alpha_n, 0, \ldots) f_i(\alpha_i)
		= 0
	\]
	וקיבלנו סתירה. \\
	כעת כמו בהוכחת הלמה הקודמת נבחר $M \triangleleft A$ מקסימלי ונסתכל על השיכון,
	\[
		K
		\hookrightarrow K[\bar{x}]
		\to A
		\to A / M
		= L
	\]
	ב־$L$ לכל $f \in K[x] \setminus K$ יש שורש כנדרש.
\end{proof}
נעבור להוכחת המשפט עצמו.
\begin{proof}[הוכחת המשפט]

	נבנה הרחבה $L_0 \subset L_1$ כך שלכל פולינום עם מקדמים ב־$L_0$ יש שורש ב־$L_1$.
	נמשיך כך ונבנה סדרת שדות,
	\[
		K = L_0 \subset L_1 \subset \cdots
	\]
	ונגדיר את $L = \bigcup_{i = 0}^\infty L_i$, זהו שדה שכן כל $f \in L[x] \setminus L$ נמצא באיזשהו $L_i[x]$ ולכן יש שורש ב־$L_{i + 1}[x]$,
	כלומר $L$ הוא שדה סגור אלגברית כפי שרצינו.
\end{proof}

\end{remark}

\end{remark}

\subsection{מודולים}
זוהי הכללה של מושג המרחב הווקטורי מעל שדה.
יהי $A$ חוג כלשהו.
\begin{definition}[מודול שמאלי]
	מודול (שמאלי) מעל $A$ הוא קבוצה $M$ ביחד עם שתי פעולות $+ : M \times M \to M$ (חיבור) ו־$\cdot : A \times M \to M$ (כפל בסקלר) המקיימות:
	\begin{enumerate}
		\item $(M, +)$ היא חבורה אבלית
		\item אסוציאטיביות: $(a b) x = a (b x)$ לכל $a, b \in A, x \in M$
		\item $1 \cdot x = x$ לכל $x \in M$
		\item פילוג: $(a + b) x = a x + b x$ וכן $a (x + y) = a x + b y$ לכל $a, b \in A, x, y \in M$
	\end{enumerate}
\end{definition}
\begin{exercise}
	הראו שהפילוג מתקיים גם עם חיסור, וכן ש־$0 \cdot x = a \cdot 0 = 0$. \\
	הראו שגם תכונת האבליות של $(M, +)$ נובעת משאר הדרישות.
\end{exercise}
\begin{example}
	הבאים הם מודולים:
	\begin{enumerate}
		\item אם $\FF$ שדה, אז כל מודול מעל $\FF$ הוא למעשה מרחב וקטורי.
		\item אם $A$ חילופי ו־$(B, \varphi : A \to B)$ אלגברה מעל $A$, אז נגדיר כפל בסקלר על־ידי $a . x = \varphi(a) x$ לכל $a \in A, x \in B$, אז $B$ מקבל מבנה של מודול מעל $A$.
		\item אם $A = \ZZ$ ו־$(M, +)$ חבורה חילופית, אז נוכל להגדיר $n \cdot x$ באופן הסטנדרטי עבור $n \in \ZZ$, אז נקבל ש־$M$ הוא מודול מעל $\ZZ$.
			משמעות הדבר היא שקבוצה היא מודול מעל $\ZZ$ אם ורק אם היא חבורה חיבורית.
		\item עבור $A$ חוג כללי ו־$I \triangleleft A$ אידאל מתקיים $a x \in I$ לכל $a \in A, x \in I$.
			לכן $I$ הוא מודול מעל $A$ עם הכפל הרגיל מצומצם ל־$I$.
		\item עבור $\FF$ שדה ו־$A = M_n(\FF)$ נגדיר את $A \times \FF^n \to \FF^n$ על־ידי כפל מטריצה בווקטור עמודה, לכן $\FF^n$ הוא מודול מעל $M_n(\FF)$.
	\end{enumerate}
\end{example}
\begin{remark}
	ניתן להגדיר גם מודול ימני שבו הכפל בסקלר הוא מימין, במקרה זה הקיבוץ יהיה בצורה,
	\[
		(x a) b
		= x (a b)
	\]
	מעל חוג חילופי אלו הן הגדרות שקולות לחלוטין, אבל בחוג כללי זהו לא המצב.
	לכן מעתה נניח שהכוונה במודול למודול שמאלי.
\end{remark}
\begin{definition}[תת־מודול]
	יהי $A$ חוג ויהי $M$ מודול מעליו.
	תת־קבוצה $N \subset M$ נקראת תת־מודול (submodule) אם $(N, +) \le (M, +)$ תת־חבורה חיבורית, ו־$N$ סגורה ביחס לכפל בסקלר, כלומר לכל $a \in A, x \in N$ מתקיים $a x \in N$.
\end{definition}
\begin{proposition}
	קבוצה היא תת־מודול אם ורק אם:
	\begin{enumerate}
		\item $N \ne \varnothing$
		\item $N + N \subset N$
		\item $a N \subset N$ לכל $a \in A$
	\end{enumerate}
\end{proposition}
\begin{example}
	אם נחשוב על $A$ כמודול מעל עצמו ונניח שהוא חילופי, אז תת־קבוצה ב־$A$ היא תת־מודול אם ורק אם היא אידאל.
\end{example}
\begin{proposition}[סכום של תתי־מודולים]
	יהיו $N_i \subset M$ עבור $i \in I$ תתי־מודולים, אז,
	\[
		N
		= \sum_{i \in I} N_i
		= \left\{ \sum_{i \in I} x_i \mid x_i \in N_i, x_i \overset{\text{a.s.}}{=} 0 \right\}
	\]
	הוא תת־מודול.
\end{proposition}
\begin{definition}[תת־מודול נפרש]
	תהי $S \subset M$ תת־קבוצה.
	נגדיר,
	\[
		\langle S \rangle
		= \sum_{s \in S} A s
		= \left\{ \sum_{i = 1}^k a_i s_i \mid a_i \in A, s_i \in S, n \in \NN \right\}
	\]
	את $\langle S \rangle \subset M$ נכנה תת־המודול הנפרש (נוצר) על־ידי $S$.
\end{definition}
\begin{definition}[כפל של מודול באידאל]
	אם $I \triangleleft A$ ו־$M$ מודול מעל $A$ אז נגדיר,
	\[
		I M
		= \left\{ \sum_{i = 1}^n a_i x_i \mid a_i \in I, x_i \in M, n \in \NN \right\}
	\]
	הוא תת־מודול ב־$M$.
\end{definition}
\begin{definition}[מודול מנה]
	יהי $M$ מודול מעל $A$ ויהי $N \subset M$ תת־מודול.
	נגדיר על חבורת המנה $M / N$ את הפעולות,
	\[
		a (x + N)
		= a x + N
	\]
	ל־$(M / N, +, \cdot)$ יש מבנה של מודול מעל $A$ ונקרא לו \textbf{מודול המנה}.
\end{definition}
\begin{proposition}
	הכפל בסקלר מוגדר היטב.
\end{proposition}
\begin{proof}
	אם $x + N = y + N$ אז $x - y \in N$ ואז $a x - a y = a (x - y) \in N$ ולכן $a x + N = a y + N$.
\end{proof}
\begin{definition}[הומומורפיזם מודולים]
	יהיו $M, N$ מודולים מעל $A$.
	הומומורפיזם $\varphi : M \to N$ הוא הומומורפיזם חבורות שבנוסף שומר על פעולת הכפל בסקלר, כלומר,
	\[
		\forall a \in A, x \in M,\ 
		\varphi(a x) = a \varphi(x)
	\]
\end{definition}
\begin{proposition}
	תכונות הומומורפיזם מודולים:
	\begin{enumerate}
		\item ברכבת הומומורפיזמים היא הומומורפיזם, כלומר אם $M \xrightarrow{\varphi} N \xrightarrow{\psi} K$ אז גם $\psi \circ \varphi$ הומומורפיזם.
		\item $\id : M \to M$ הוא הומומורפיזם
		\item אם $\varphi : M \to N$ איזומורפיזם (הומומורפיזם הפיך) אז גם $\varphi^{-1} : N \to M$ הומומורפיזם.
		\item אם $\varphi, \psi : M \to N$ הומומורפיזמים, אז גם $\varphi \pm \psi : M \to N$ הומומורפיזם.
		\item אם $\varphi \in \Hom(M, N)$ (שנגדיר מיד) אז $\ker \varphi = \{ x \in M \mid \varphi(x) = 0 \}$ היא תת־מודול, התמונה $\varphi(M) \subset N$ היא גם תת־מודול.
	\end{enumerate}
\end{proposition}
\begin{definition}[חבורת ההומומורפיזמים]
	נסמן את,
	\[
		\Hom(M, N)
		= \Hom_A(M, N)
		= \{ \varphi : M \to N \text{ homomorphism} \}
	\]
	היא חבורה חילופית.
\end{definition}
\begin{proposition}
	אם $A$ חילופי אז $\Hom(M, N)$ הוא מודול עם 
\end{proposition}
\begin{proof}
	נראה שלכל סקלר גם $a \varphi$ הומומורפיזם,
	\[
		(a \varphi)(b x)
		= a \varphi(b x)
		= a b \varphi(x)
		= b a \varphi(x)
		= b ((a \varphi)(x))
	\]
	ולכן $a \varphi$ הומומורפיזם.
\end{proof}
גם משפטי האיזומורפיזם תקפים עבור מודולים.
\begin{theorem}[האיזומורפיזם הראשון]
	יהי $\varphi \in \Hom(M, N)$, אז $\varphi(M) \simeq M / \ker \varphi$ וישנה ההתאמה $\varphi(x) \leftrightarrow x + \ker \varphi$.
\end{theorem}
\begin{theorem}[ההתאמה]
	אם $M$ מודול ו־$N \subset M$ תת־מודול, נסמן $\pi : M \to M / N$ העתקת המנה, אז קיימת התאמה בין $\{ N \subset P \subset M \text{ submodules} \}$ לבין $\{ M / N \text{ submodules} \}$,
	על־ידי $P \mapsto \pi(P) = P / N$ ו־$Q \mapsto \pi^{-1}(Q)$.
\end{theorem}
\begin{theorem}[האיזומורפיזם השני]
	יהי $M$ מודול ויהיו $N, P \subset M$ תת־מודולים, אז מתקיים,
	\[
		(N + P) / P
		\simeq N / (N \cap P)
	\]
	על־ידי ההתאמה $x + P \leftrightarrow x + (N \cap P)$/
\end{theorem}
\begin{theorem}[האיזומורפיזם השלישי]
	אם $P \subset N \subset M$ תתי־מודולים, אז,
	\[
		M / N
		\simeq (M / P) / (N / P)
	\]
	נתון על־ידי $x + N \leftrightarrow (x \mod P) + (N / P)$.
\end{theorem}
\begin{definition}[סכום ישר חיצוני]
	יהי $A$ חוג ונגדיר מודולים מעליו ${\{ M_i \}}_{i \in I}$.
	\textbf{סכום ישר חיצוני} מוגדר על־ידי,
	\[
		\bigoplus_{i \in I} M_i
		= \left\{ f : I \to \bigcup M_i \mid \forall i \in I,\ f(i) \in M_i, f(i) \overset{\text{a.s.}}{=} 0 \right\}
	\]
	עם הפעולות $(f + g)(i) = f(i) + g(i)$ והפעולה $(a f)(i) = a f(i)$.
	במקרה זה $\bigoplus_I M_i$ הוא $A$־מודול.
\end{definition}
\begin{definition}[סכום ישר פנימי]
	אם $M$ מודול, $M_i \subset M$ אוסף של תתי־מודולים, נאמר ש־$M = \bigoplus_{i \in I} M_i$ אם כל $x \in M$ ניתן לרשום ביחידות על־ידי,
	\[
		x
		= \sum_{i \in I} x_i
	\]
	כאשר $x_i \in M_i$ ולכמעט כל $i$ גם $x_i = 0$.
\end{definition}
\begin{proposition}
	הגדרה זו שקולה לתנאים ש־$M = \sum_I M_i$ ובנוסף ש־$M_i \cap \sum_{j \ne i} M_j = \{ 0 \}$ לכל $i$.
\end{proposition}
\begin{exercise}
	הוכיחו טענה זו.
\end{exercise}
סכום ישר פנימי איזומורפי לסכום ישר חיצוני של $M_i$ על־ידי,
\[
	f(i) = x_i
	\leftrightarrow
	x = \sum_I x_i
\]
\begin{definition}[מכפלה ישרה]
	נגדיר מכפלה ישרה (חיצונית) על־ידי,
	\[
		\prod_{i \in I} M_i
		= \left\{ f : I \to \bigcup_I M_i \mid f(i) \in M_i \right\}
	\]
	אם $I$ סופי אז $\prod M_i = \bigoplus M_i$, אחרת $\prod M_i$ הוא אובייקט יותר גדול.
\end{definition}
\begin{definition}[מודול חופשי בתיאור חיצוני]
	\[
		A^{\oplus I}
		= \bigoplus_{i \in I} A
	\]
	נקרא \textbf{המודול החופשי} מעל $A$ של אוסף היוצרים ב־$I$. \\
	לכל $i \in I$ נתאים וקטור בסיס סטנדרטי,
	\[
		e_i
		= j \mapsto
		\begin{cases}
			1 \in A & j = i \\
			0 \in A & j \ne i
		\end{cases}
	\]
\end{definition}
\begin{definition}[בסיס]

\end{proof}
\begin{definition}[מודול נוצר על־ידי יוצרים ויחסים]

\end{theorem}
\begin{proof}
	קיום של $\varphi$:
	$b_i$ פורשים את $M$ שכן $e_i$ פורשים את $A^{\oplus I}$.
	לכן כל $y \in M$ ניתן להציג על־ידי $y = \sum a_i b_i$ עבור $a_i \in A$.
	נגדיר $\varphi(y) = \sum a_i x_i \in N$, ונראה ש־$\varphi$ מוגדר היטב.

	לכן $\sum a_i x_i = \sum a_i' x_i$ ו־$\varphi$ מוגדר היטב. \\
	קל להראות ש־$\varphi$ הומומורפיזם, ויחידות נובעת מכך ש־$b_i$ פורשים את $M$.
\end{proof}

\begin{definition}[העתקה בילינארית מעל חוג]
	יהיו $M, N, P$ מודולים מעל $A$.
	פונקציה $B : M \times N \to P$ נקראת \textbf{בילינארית} אם מתקיים,
	\[
		B(a x + b x', y)
		= a B(x, y) + b B(x', y)
		\qquad
		B(x, a y + b y')
		= a B(x, y) + b B(x, y')
	\]
\end{definition}
\begin{remark}
	אם $\psi : P \to Q$ הומומורפיזם אז $\psi \circ B : M \times N \to Q$ גם בילינארית.
\end{remark}
המטרה שלנו היא לבנות העתקה בילינארית אוניברסלית $\otimes : M \times N \to P$ כך שלכל $B : M \times N \to Q$ בילינארית היא מהצורה $B(x, y) = \psi(x \otimes y)$ כאשר $\psi \in \Hom(P, Q)$.
\begin{definition}[מכפלה טנזורית]
	יהיו $M, N$ מודולים מעל $A$, ונגדיר אוסף יחסים על $A^{\oplus(M \times N)}$ על־ידי,
	\[
		R
		= \{ e_{a x + b x', y} - a e_{x, y} - b e_{x', y}, e_{x, a y + b y'} - a e_{x, y} - b e_{x, y'} \mid x, x' \in M, y, y' \in N, a, b \in A \}
	\]

\end{definition}

	\[
		x \otimes (a y + b y')
		= a x \otimes y + b x \otimes y',
		\qquad
		(a x + b x') \otimes y
		= a x \otimes y + b x' \otimes y
	\]
\end{proposition}

		x \otimes y
		= \psi(e_{x, y} \mod \langle R \rangle)

	\]
\end{proof}

	\]

		\item חוק פילוג בכיוון ההפוך,

			\]
		\item $(M \otimes N) \otimes P \cong M \otimes (N \otimes P)$ מתקבל על־ידי $(x \otimes y) \otimes z \mapsto x \otimes (y \otimes z)$
	\end{enumerate}
\end{proposition}
\begin{proof}
	נראה את הוכחת התכונה הראשונה, השאר דומות. \\
	$\alpha$ היא בילינארית ולכן מוגדרת היטב, וכן זוהי ההעתקה היחידה המקיימת את התכונה הזאת.

		= (0, y) \otimes z

		= \beta(w) + \gamma(u)
	\]

\begin{definition}[סדרה מדויקת]

	\]

\end{definition}
\begin{notation}

		\xrightarrow{\psi} P
		\to 0
	\]

		= \im \psi
		= N / \ker \psi
		= N / \im \varphi
	\]

	\[
		0
		\to M
		\hookrightarrow N
		\to N / M
		\to 0
	\]
\end{notation}

	\[
		(\psi \otimes \id_Q)(y \otimes t)
		= \psi(y) \otimes \id_Q(t)
		= \theta(y) \otimes t
		= x \otimes t
	\]
	אבל $x \otimes t \in \im(\psi \otimes \id_Q)$ ולכן $P \otimes Q = \im(\psi \otimes \id_Q)$.

	נעבור להראות $\im(\varphi \otimes \id) = \ker(\psi \otimes \id)$.
	מתקיים,
	\[
		(\psi \otimes \id) \circ (\varphi \otimes \id)
		= (\psi \circ \varphi) \otimes (\id \otimes \id)
		= 0
	\]
	כאשר $\psi \otimes \varphi = 0$ מההנחה שלנו.
	אז $\im(\varphi \otimes \id) \subset \ker(\psi \otimes \id)$, נשאר להראות שיש שוויון (ונשאר כתרגיל להראות שמכפלה טנזורית מתחלפת עם הרכבה).
	$\psi$ על ולכן לכל $x \in P$ נבחר $f(x) \in N$ כך ש־$\psi(f(x)) = x$ ולכן $\psi \circ f = \id$, נגדיר,
	\[
		B : P \times Q \to (N \otimes Q) / \im(\varphi \otimes \id),
		\qquad
		B(x, t)
		= f(x) \otimes t + \im(\varphi \otimes \id)
	\]
	$B$ בילינארית ולכן,
	\[
		f(x) \otimes (a t + b s)
		= a f(x) \otimes t + b f(x) \otimes s
	\]
	וכן,
	\[
		f(a x + b y) \otimes t - a f(x) \otimes t - b f(y) \otimes t
		= (f(a x + b y) - a f(x) - b f(y)) \otimes t
		\in \im(\varphi \otimes t)
	\]
	שכן $f(a x + b y) \otimes t - a f(x) \otimes t - b f(y) \otimes t \in \ker \psi = \im \varphi$.
	לכן $B(a x + b y, t) - a B(x, t) - b B(y, t) = 0$ ונסיק ש־$B$ בילינארית.
	לכן קיים ויחיד $\lambda \in \Hom(P \otimes Q, (N \otimes Q) / \im(\varphi \otimes \id))$ כך שמתקיים,
	\[
		\lambda(x \otimes b)
		= f(x) \otimes t + \im(\varphi \otimes \id)
	\]
	אם $x \in N$ אז,
	\[
		\psi(x - f(\psi(x)))
		= \psi(x) - \psi(f(\psi(x)))
		= \psi(x) - \psi(x)
		= 0
	\]
	כלומר $x - f(\psi(x)) \in \ker \psi = \im \varphi$.
	נובע,
	\[
		\lambda(\psi(x) \otimes t)
		\equiv f(\psi(x)) \otimes t
		\equiv x \otimes t
		\mod \im(\varphi \otimes \id)
	\]
	ולכן $x \equiv f(\psi(x)) \mod \im \varphi$.
	קיבלנו שמתקיים $\lambda \circ (\psi \otimes \id)$ היא העתקת המנה $N \otimes Q \to (N \otimes Q) / \im(\varphi \otimes \id)$.
	כעת אם $u \in \ker(\psi \otimes \id)$ אז $\lambda((\psi \otimes \id)(u)) = 0$ ולכן $u \in \im(\varphi \otimes \id)$, כלומר קיבלנו את הנדרש.
\end{proof}
\begin{corollary}
	יהי $I \triangleleft A$, אז $M \otimes A / I \cong M / I M$ מתקבל על־ידי $x \otimes (a \mod I) \mapsto a x \mod I M$.
\end{corollary}
\begin{proof}
	$0 \to I \overset{\iota}{\hookrightarrow} A \xrightarrow{\pi} A / I \to 0$ מדויקת.
	ממשפט הדיוק מימין נקבל שגם,
	\[
		I \otimes M
		\to A \otimes M
		\to (A / I) \otimes M
		\to 0
	\]
	מדויקת, וכן נזכור כי $A \otimes M = M$ (כמודולים איזומורפיים) ולכן נקבל $\im(\iota \otimes \id_M) = I M$ ולכן,
	\[
		(A / I) \otimes M
		\cong (A \otimes M) / \ker(\pi \otimes \id)
		= (A \otimes M) / \im(\iota \otimes \id)
		= M / I M
	\]
	וצריך לבדוק שאכן ההעתקה שהגדרנו מקיימת את הדרוש.
\end{proof}
נראה כי אין דיוק משמאל עם הדוגמה הבאה.
\begin{example}
	נסמן $A = \ZZ$ ונגדיר את הסדרה המדויקת,
	\[
		0
		\to 2 \ZZ
		\hookrightarrow \ZZ
		\to \ZZ / 2 \ZZ
	\]
	נכפול טנזורית עם $\ZZ / 2 \ZZ$ ונקבל,
	\[
		0
		\to 2 \ZZ / 4 \ZZ
		\xrightarrow{0} \ZZ / 2 \ZZ
		\xrightarrow{\id} \ZZ / 2 \ZZ
	\]
	ואין כאן דיוק.
\end{example}

\subsection{בניות על־ידי מכפלות טנזוריות}
נניח ש־$A$ הוא חוג חילופי, ו־$(B, f : A \to B)$ אלגברה על $A$.
נוכל לצמצם סקלרים מ־$B$ ל־$A$:
אם $M$ הוא מודול מעל $B$ נהפוך אותו ל־$A$־מודול על־ידי כפל בסקלר,
\[
	a x
	= f(a) x
	\in M
\]
לכל $a \in A, x \in M$.
במקרה זה,
\[
	(a b) x
	= f(a b) x
	= f(a) f(b) x
	= f(a) b x
	= a (b x)
\]
ושאר האקסיומות נשמרות באופן דומה.

נוכל גם להרחיב סקלרים:
אם $M$ הוא מודול מעל $A$ ו־$B$ הוא מודול מעל $A$ עם הפעולה $a b = f(a) b$ לכל $a \in A, b \in B$.
נגדיר,
\[
	M_B
	= B \otimes_A M
\]
אז נקבל ש־$M_B$ הוא מודל מעל $A$.
נהפוך אותו ל־$B$־מודול על־ידי,
\[
	b \sum b_i \otimes x_i
	= \sum b b_i \otimes
	\in M_B
\]
כדי לבדוק שהפעולה מוגדרת היטב עלינו להראות ש־$b, x \mapsto b x$ בילינארית (תרגיל), קל לבדוק שמתקיימות אקסיומות המודול.
\begin{definition}[הרחבת סקלרים]
	$M_B$ הוא $B$־מודול שנקרא \textbf{הרחבת הסקלרים} מ־$A$ ל־$B$.
\end{definition}
\begin{example}
	\begin{enumerate}
		\item $M = A^{\otimes I}$ חופשי, אז,
			\[
				M_B
				= A^{\otimes I} \otimes_A B
				= {(A \otimes_A B)}^{\otimes I}
				= B^{\otimes I}
			\]
		\item אם $I \triangleleft A$ ו־$B = A / I$ אז,
			\[
				M_B
				= (A / I) \otimes M
				\cong M / I M
			\]
	\end{enumerate}
\end{example}
\begin{exercise}
	אם $M, N$ מודולים מעל $A$ ו־$\varphi \in \Hom_A(M, N)$ אז $\varphi_B : = \id_B \otimes \varphi : M_B \to N_B$ הומומורפיזם של $B$־מודולים,
	\[
		{( \id_M )}_B
		= \id_{M_B},
		\qquad
		{(\varphi \circ \psi)}_B
		= \varphi_B \circ \psi_B
	\]
\end{exercise}
\begin{theorem}[תכונת האוניברסליות של הרחבת סקלרים]
	יהי $M$ מודול מעל $A$, ו־$N$ מודול מעל $B$, $\varphi \in \Hom_A(M, N)$.
	אז לכל $f : A \to B$ קיים ויחיד $\psi \in \Hom_B(M_B, N)$ כך שמתקיים $\psi \circ (f \otimes \id_M) = \varphi$, וכן $\psi(b \otimes x) = b \varphi(x)$.
\end{theorem}
\begin{definition}[מכפלה טנזורית של אלגברות]
	יהי $A$ חוג, $(B, \varphi : A \to B)$ ו־$(C, \psi : A \to C)$ שתי אלגברות מעל $A$.
	בפרט $B, C$ הם מודולים מעל $A$ ולכן ניתן להגדיר את $A \otimes_A C$, על מבנה זה נגדיר מבנה של אלגברה על־ידי,
	\[
		\left(\sum b_i \otimes c_i\right)\left(\sum b_j' \otimes c_j'\right)
		= \sum_{i, j} b_i b_j' \otimes c_i c_j'
	\]
\end{definition}
הכפל מוגדר היטב שכן ההעתקה $b, c \mapsto b b' \otimes c c'$ בילינארית ולכן הכפל לא תלוי בהצגת האיבר הראשון כסכום טנזורים פשוטים.
אותה הטענה נכונה גם לערך השני.
\begin{exercise}
	בדקו ש־$B \otimes_A C$ חוג חילופי.
\end{exercise}
נגדיר $f : A \to B \otimes_A C$ על־ידי,
\[
	f(a)
	= \varphi(a) \otimes 1
	= a 1 \otimes 1
	= 1 \otimes a 1
	= 1 \otimes \psi(a)
\]
\begin{exercise}
	בדקו ש־$f$ הומומורפיזם של חוגים.
\end{exercise}
זה מוביל אותנו להגדרה הבאה,
\begin{definition}[מכפלה טנזורית של אלגברות]
	$(B \otimes_A C, f)$ היא אלגברה מעל $A$, נקרא לה \textbf{המכפלה הטנזורית} של $(B, \varphi)$ ו־$(C, \psi)$.
\end{definition}
\begin{example}
	יהי $A$ חוג ויהיו $B = A[\bar{x}], C = A[\bar{x}]$ עבור $\bar{x}, \bar{y}$ אוספי משתנים.
	אז,
	\[
		B \otimes_A C
		\cong A[\bar{x}, \bar{u}]
	\]
	זאת שכן $A[\bar{x}] = A^{\oplus \Mon(\bar{x})}$ כ־$A$־מודולו.
	באופן דומה נקבל גם $A[\bar{y}] = A^{\oplus \Mon(\bar{y})}$, לכן,
	\[
		A[\bar{x}] \otimes_A A[\bar{y}]
		\cong A^{\oplus (\Mon(\bar{x}) \times \Mon(\bar{y}))}
	\]
	אבל ניתן לזהות,
	\[
		\Mon(\bar{x} \cup \bar{y})
		= \Mon(\bar{x}) \times \Mon(\bar{y})
	\]
	כאשר נזהה $(m(\bar{x}), n(\bar{y})) \leftrightarrow m(\bar{x}) n(\bar{y})$.
	ניתן לבדוק שפעולות הכפל ב־$A[\bar{x} \cup \bar{y}]$ ו־$A^{\oplus \Mon(\bar{x}) \times \Mon(\bar{y})}$ מתאימים.
\end{example}
\begin{example}
	כמקרה פרטי של הדוגמה הקודמת,
	\[
		A[x] \otimes A[y]
		= A[x, y]
	\]
\end{example}
באופן כללי נקבל $f(\bar{x}) \otimes g(\bar{x}) \leftrightarrow f(\bar{x}) g(\bar{y})$.
\begin{definition}[חוג הקורדינטות של קבוצה אלגברית]
	יהי $K$ שדה סגור אלגברית, ו־$\bar{x} = (x_1, \ldots, x_n)$.
	ניזכר בהגדרה,
	\[
		V
		= Z(S)
		= \{ a \in K^n \mid \forall f \in S,\ f(a) = 0 \}
		\subset K^n
	\]
	וכן,
	\[
		I(V)
		= \{ f \in K[\bar{x}] \mid \forall a \in V,\ f(a) = 0 \}
	\]
	אז נגדיר את,
	\[
		K[V]
		= \{ f |_V \mid f \in K[\bar{x}] \}
	\]
	כאוסף הפונקציות $K^n \to K$.
	קבוצה זו נקראת \textbf{חוג הקורדינטות} של $V$.
\end{definition}
\begin{remark}
	$f - g \in I(V) \iff f |_V = g |_V$ ולכן ניתן לזהות $K[V] \cong K[\bar{x}] / I(V)$ על־ידי $f |_V \leftrightarrow (f \mod I(V))$.
\end{remark}
תהי $W \subset K^m$ קבוצה אלגברית נוספת.
נבחן את $V \times W \subset K^{n + m}$, גם $V \times W$ אלגברית ומוגדרת על־ידי $I(V) \cup I(W)$,
כאשר משכנים $K[\bar{x}], K[\bar{y}] \subset K[\bar{x}, \bar{y}]$.
\begin{proposition}
	מתקיים,
	\[
		K[V \times W]
		\cong K[V] \otimes_K K[W]
	\]
	כאיזומורפיזם של $K$־אלגברות.
\end{proposition}
\begin{proof}
	נגדיר את $\varphi : K[V] \otimes K[W] \to K[V \times W]$ על־ידי,
	\[
		\varphi(\sum_i f_i |_V \otimes g_i |_W)
		= \sum_i f_i(\bar{x}) g_i(\bar{y}) |_{V \times W}
	\]
	יש לבדוק שהפעולה $f |_V, g |_W \mapsto f(\bar{x}) g(\bar{y}) |_{V \times W}$ מוגדרת היטב ובילינארית.
	לכן $\varphi$ מוגדר היטב, קל לבדוק ש־$\varphi$ הוא הומומורפיזם של $K$־אלגברות ונשאר להוכיח שהוא חד־חד ערכי ועל. \\
	כל פולינום ב־$K[\bar{x}, \bar{y}]$ ניתן להציג על־ידי $\sum f_i(\bar{x}) g_i(\bar{y})$ ולכן $\varphi$ על.

	נעבור לחד־חד ערכיות.
	יהי $\alpha = \sum_{i = l}^l f_i |_V \otimes g_i |_W \in \ker \varphi$ כך ש־$l$ מינימלי.
	נסמן גם $\bar{f}_i = f_i |_V, \bar{g}_i = g_i |_W$.
	אז $\bar{g}_1, \ldots, \bar{g}_l$ בלתי־תלויים לינארית מעל $K$, אחרת נוכל להציג $\bar{g}_l = \sum_{i < l} \alpha_i \bar{g}_i$ ואז,
	\[
		\sum_1^l \bar{f}_i \otimes \bar{g}_i
		= \sum_1^{l - 1} (\bar{f}_i + a_i) \otimes \bar{g}_i
	\]
	וקיבלנו הצגה יותר קצרה בסתירה למינימליות של $l$. \\
	לכל $a \in V$ נבחן את ההומומורפיזם,
	\[
		s_a : K[V] \to K,
		\qquad
		s_a(\bar{f}) = f(a)
	\]
	הומומורפיזם ההצבה.
	על־ידי מכפלה טנזורית עם $\id_{K[W]}$ נקבל,
	\[
		(s_a \otimes \id_{K[W]})(\sum_{i = 1}^l \bar{f}_i \otimes \bar{g}_i)
		= \sum_{i = 1}^l f_i(a) \bar{g}_i
		\equiv 0
	\]
	כי $\sum \bar{f}_i \otimes \bar{g}_i \in \ker \varphi$ ולכן $\sum_1^l f_i g_i \in I(V \times W)$.
	מכיוון ש־$\bar{g}_i$ בלתי־תלויים נסיק ש־$f_i(a) = 0$ לכל $i$ ולכל $a \in V$, כלומר $\bar{f}_i \equiv 0$ ו־$\alpha = \sum \bar{f}_i \otimes \bar{g}_i = 0$, כלומר $\varphi$ חד־חד ערכי.
\end{proof}
\begin{theorem}[התכונה האוניברסלית של מכפלה טנזורית של אלגברות]
	נגדיר הומומורפיזמים $i_B : B \to B \otimes_A C$ וכן $i_C : C \to B \otimes_A C$ על־ידי $i_B(b) = b \otimes 1_C$ ו־$i_C(c) = 1_B \otimes c$.
	$i_B, i_C$ הומומורפיזמים של $A$־אלגברות. \\
	כעת נניח ש־$f : B \to D, g : C \to D$ זוג הומומורפיזמים של $A$־אלגברות.
	אז קיים ויחיד הומומורפיזם $\varphi : B \otimes_A C \to D$ המקיים,
	\[
		\varphi \circ i_B = f,
		\qquad
		\varphi \circ I_C = g
	\]
	והוא מתקבל על־ידי $\varphi(b \otimes c) = f(b) g(c)$.
\end{theorem}
$b, c \mapsto f(b) g(c)$ בילינארית ולכן $\varphi$ מוגדר היטב, וקל לבדוק שהוא הומומורפיזם של $A$־אלגברות.
\begin{remark}
	 גם למכפלה $B \times C$ יש תכונה אוניברסלית דואלית לתכונה של מכפלה טנזורית.
\end{remark}

\subsection{מיקום}
באנגלית localization.
באופן כללי בחוג $A$ אין חילוק, אנו נרצה לבנות חוג אחר ''גדול יותר'' שבו ניתן לחלק בחלק מהאיברים ב־$A$.
\begin{definition}[תת־קבוצה כפלית]
	יהי $A$ חוג, תת־קבוצה $S \subset A$ נקראת \textbf{כפלית} (multiplicative) אם מתקיימות התכונות:
	\begin{enumerate}
		\item $1 \in S$
		\item סגירות לכפל: $S \cdot S \subset S$, כלומר לכל $s, t \in S$ מתקיים $s t \in S$
	\end{enumerate}
\end{definition}
\begin{example}
	הבאות הן תת־קבוצות כפליות:
	\begin{enumerate}
		\item $A^\times$
		\item אם $f \in A$ אז $\{1, f, f^2, \ldots\}$
		\item אם $P \triangleleft A$ ראשוני, אז $A \setminus P$
	\end{enumerate}
\end{example}
יהי $A$ חוג ותהי $S \subset A$ כפלית, נבנה אלגברה מעל $A$ שבה כל האיברים ב־$S$ הפיכים כך שהיא גם תהיה אוניברסלית ביחס לתכונה זו.
נגדיר יחס שקילות $\sim$ על $A \times S$ על־ידי,
\[
	(a, s) \sim (b, t)
	\iff \exists r \in S,\ r(a t - b s) = 0
\]
\begin{proposition}
	$\sim$ אכן יחס שקילות.
\end{proposition}
\begin{proof}
	$(a, s) \sim (a, s)$ כי $1 \cdot (a s - s a) = 0$, ולכן קיבלנו רפלקסיביות.

	אם $(a, s) \sim (b, t)$ אז קיים $r \in S$ כך ש־$r (a t - b s) = 0$ אבל אז נקבל שגם $r (b s - a t) = 0$, כלומר קיבלנו סימטריה.

	נעבור לטרנזיטיביות.
	נניח ש־$(a, s) \sim (b, t) \sim (c, u)$ ולכן קיימים $r, r' \in S$ כך שמתקיים,
	\[
		r (a t - b s) = 0,
		\qquad
		r' (v u - c t) = 0
	\]
	בהתאם מתקיים,
	\[
		(r r' t) (a n - c s)
		= r' a r (a t - b s) + r s r' (b u - c t)
		= 0
	\]
	ולכן גם $(a, s) \sim (c, u)$.
\end{proof}
\begin{notation}
	נסמן ב־$\frac{a}{s}$ את מחלקת השקילות של $(a, s)$, לכן,
	\[
		\frac{a}{s} = \frac{b}{t}
		\iff \exists r \in S,\ r (a t - b s) = 0
	\]
	ונסמן $S^{-1} A = \{ \frac{a}{s} \mid a \in A, s \in S \}$.
\end{notation}
נרצה לתת ל־$S^{-1} A$ מבנה של חוג ולכן נגדיר,
\[
	\frac{a}{s} + \frac{b}{t}
	= \frac{a t + b s}{s t},
	\qquad
	\frac{a}{s} \cdot \frac{b}{t}
	= \frac{a b}{s t}
\]
\begin{proposition}
	מתקיים:
	\begin{enumerate}
		\item שתי הפעולות מוגדרות היטב.
		\item $(S^{-1} A, +, \cdot)$ חוג חילופי, כאשר $0_{S^{-1} A} = \frac{0}{1}$ וגם $1_{S^{-1} A} = \frac{1}{1}$
		\item ההעתקה $\loc : A \to S^{-1} A$ המוגדרת על־ידי $\loc(a) = \frac{a}{1}$ היא הומומורפיזם, ונקרא לה העתקת המיקום ב־$S$
		\item $\loc$ שולחת כל איבר ב־$S$ לאיבר הפיך, כלומר $\frac{1}{s} \cdot \frac{s}{1} = \frac{1}{1} = 1_{S^{-1} A}$ ומתקיים,
			\[
				{(S^{-1} A)}^\times
				= \left\{ \frac{a}{s} \mid \exists b \in A,\ b a \in S \right\}
			\]
	\end{enumerate}
\end{proposition}
\begin{proof}
	\textbf{1.}
	נבדוק עבור כפל, כלומר נבדוק שאם $\frac{a}{s} = \frac{a'}{s'}$ וכן $r(a s' - a' s) = 0$ ולכן,
	\[
		r (a s' b t - a' s b t) = 0
	\]
	ומכאן נובע שמתקיים,
	\[
		\frac{a b}{s t}
		= \frac{a' b}{s' t}
	\]
	ונובע שהכפל מוגדר היטב, הבדיקה עבור חיבור היא דומה.
\end{proof}
אז $(S^{-1} A, \loc)$ היא אלגברה על $A$ שבה על האיברים ב־$S$ הם הפיכים.

\subsection{מיקום --- המשך}
בשיעור הקודם הוכחנו שבהינתן קבוצה כפלית בחוג חילופי $S \subset A$ אז ניתן להגדיר חוג חדש.
\begin{definition}[מיקום של חוג]
	$(S^{-1} A, +, \cdot)$ הוא חוג ונקרא לו \textbf{המיקום} של $A$ ב־$S$.
\end{definition}
\begin{definition}[העתקת המיקום]
	$\loc : A \to S^{-1} A$ המוגדרת על־ידי $\loc(a) = \frac{a}{1}$ היא הומומורפיזם המיקום, אם $s \in S$ אז $\loc(s) = \frac{s}{1}$ הפיך על־ידי $\frac{1}{s}$.
\end{definition}
נעבור לדוגמות.
\begin{example}
	נראה מקרים של מיקומים מעל קבוצות כפליות.
	\begin{enumerate}
		\item אם $S = \{ 1 \}$ אז $\loc : A \xrightarrow{\sim} S^{-1} A$
		\item אם $0 \in S$ רז $S^{-1} A = 0$ כי $\frac{1}{s} = \frac{b}{t}$ לכל $a, b, s, t$ כי $0 \cdot a t = 0 \cdot b s$
		\item מיקום באיבר בודד: אם $f \in A$ ו־$S = \{1, f, f^2, \ldots\}$ אז נסמן את $A_f = S^{-1} A$ ולכן,
			\[
				A_f
				= \{ \frac{a}{f^n} \mid a \in A, n \in \NN_0 \}
			\]
		\item מיקום באידאל ראשוני:
			אם $P \triangleleft A$ אז $S = A \setminus P$ כפלית ונסמן,
			\[
				A_P
				= \{ S^{-1} P \mid \frac{a}{s} \mid a \in A, s \in A \setminus P \}
			\]
		\item שדה השברים של תחום $A$.
			$(0) \triangleleft A$ ראשוני, נגדיר את,
			\[
				\Frac(A)
				= A_{(0)}
				= \{ \frac{a}{b} \mid a \in A, 0 \ne b \in A \}
			\]
			בפרט $\Frac(\ZZ) = \QQ$. \\
			אם $K$ שדה אז נסמן $K(x_1, \ldots, x_n) = \Frac(K[x_1, \ldots, x_n])$ שדה הפונקציות הרציונליות מעל $K$ במשתנים $x_1, \ldots, x_n$
	\end{enumerate}
\end{example}
\begin{theorem}[התכונה האוניברסלית של מיקום]
	יהי $f : A \to B$ הומומורפיזם של חוגים, ותהי $S \subset A$ כפלית, נניח כי $f(S) \subset B^\times$.
	אז קיים ויחיד הומומורפיזם $\psi : S^{-1} A \to B$ כך שמתקיים,
	\[
		\loc \circ \psi
		= f
	\]
\end{theorem}
\begin{proof}
	נגדיר,
	\[
		\psi(\frac{a}{s})
		= f{(s)}^{-1} f(a)
	\]
	נראה ש־$\psi$ מוגדר היטב.
	נניח ש־$\frac{a}{s} = \frac{b}{t}$, כלומר $r a t = r b s$ לכל $r \in S$ ולכן,
	\[
		f(r) f(a) f(t)
		= f(r) f(b) f(s)
		\implies f{(s)}^{-1} f(a)
		= f{(t)}^{-1} f(b)
	\]
	על־ידי שימוש בהנחת ההפיכות.
	נסיק שההעתקה מוגדת היטב, וניתן לבדוק בדיקה ישירה שזהו אכן הומומורפיזם, לכן נעבור ליחידות. \\
	יחידות נובעת מ־$f = \psi \circ \loc$,
	\[
		f(s) \psi(\frac{a}{s})
		= \psi(\frac{s}{1}) \psi(\frac{a}{s})
		= \psi(\frac{a}{1})
		= f(a)
	\]
	ולכן $\psi(\frac{a}{s}) = f{(s)}^{-1} f(a)$.
\end{proof}
\begin{example}
	נניח ש־$K$ שדה סגור־אלגברית, ותהי $V \subset K^n$ קבוצה אלגברית.
	ראינו כי $K[V] \cong K[\bar{x}] / I(V)$, ותהי $a \in V$, נגדיר,
	\[
		S
		= \{ g \in K[V] \mid g(a) \ne 0 \}
	\]
	אז $S$ כפלית.
	נסמן,
	\[
		O_{V, a}
		= S^{-1} K[V]
		= \{ \frac{f}{g} \mid f, g \in K[V], g(a) \ne 0 \}
	\]
	אז $\frac{f}{g}$ מוגדרת רק בסביבה פתוחה זריצקי של $a$ (בסביבה בה $g \ne 0$) ולא בהכרח על כל $V$.
	נקבל גם ש־$\frac{f}{g} = \frac{h}{q}$ אם ורק אם $\frac{f}{g} = \frac{h}{q}$ על סביבה פתוחה זריצקי של $a$.
\end{example}
\begin{remark}
	$O_{V, a} \cong {K[V]}_P$ עבור $P = \{ f \in K[V] \mid f(a) = 0 \}$ ראשוני.
\end{remark}
\begin{definition}[מיקום של אידאל]
	אם $I \triangleleft A$ ו־$S \subset A$ כפלית, אז נגדיר,
	\[
		S^{-1} I
		= \{ \frac{a}{s} \mid a \in I, s \in S \}
		\triangleleft S^{-1} A
	\]
	הוא אידאל ב־$S^{-1} A$.
\end{definition}
\begin{exercise}
	הוכיחו כי זהו אכן אידאל.
\end{exercise}
\begin{proposition}
	כל אידאל ב־$S^{-1} A$ הוא מהצורה $S^{-1} I$ כאשר $I \triangleleft A$.
\end{proposition}
\begin{proof}
	יהי $J \triangleleft S^{-1} A$ ונגדיר את $I = \loc^{-1}(J) = \{ a \in A \mid \frac{a}{1} \in J \}$.
	אז $S^{-1} I = J$:
	\[
		\frac{a}{s} \in J
		\implies
		\frac{a}{1} = \frac{a}{s} \cdot \frac{s}{1} \in J
		\implies \frac{a}{s} \in S^{-1} A
	\]
	ולכן $J \subset S^{-1} I$, והכיוון השני ברור.
\end{proof}
\begin{proposition}
	יש התאמה חד־חד ערכית על ושומרת הכלה בין $\{ P \triangleleft A \text{ prime}, P \cap S = \varnothing \}$ ו־$\{ Q \triangleleft S^{-1} A \text{ prime} \}$,
	והיא מתקבלת על־ידי $Q \mapsto \loc^{-1}(Q)$ ו־$P \mapsto S^{-1} P$.
\end{proposition}
\begin{proof}
	יהי $P \triangleleft A$ כך ש־$P \cap S = \varnothing$.
	תחילה נראה ש־$\frac{a}{s} \in S^{-1} P$ אם ורק אם $a \in P$.
	נניח ש־$\frac{a}{s} \in S^{-1} A$, כלומר $\frac{a}{s} = \frac{b}{t}$ עבור $b \in P, t \in S$.
	$r a t = r b s \in P$ עבור $r \in S$, $r, t \notin P$ ולכן $a \in P$ (מראשוניות) כנדרש.

	נראה ששתי ההעתקות שהגדרנו הופכיות זו לזו ומוגדרות היטב (שולחות איברים לטווח שהגדרנו).
	יהי $P \triangleleft A$ כך ש־$P \cap S = \varnothing$, אז,
	\[
		\loc^{-1}(S^{-1} A)
		= P
	\]
	ישירות מהחלק הראשון של ההוכחה.
	נבדוק ש־$S^{-1} A$ ראשוני. אם $\frac{a}{s} \cdot \frac{b}{t} \in S^{-1} P$ אז לפי החלק הראשון שוב נקבל ש־$a b \in P$ ואז או $a \in P$ ולכן $\frac{a}{s} \in S^{-1} A$ או ש־$b \in P$ ולכן $\frac{b}{s} \in S^{-1} P$.
	אם $Q \triangleleft S^{-1} A$ ראשוני אז $P = \loc^{-1}(Q)$ ראשוני כתמונה הפוכה של ראשוני על־ידי הומומורפיזם.
	קל לבדוק ש־$S^{-1} \loc^{-1}(Q) = Q$ שכן טענה זו נכונה לכל אידאל (לא רק ראשוניים). \\
	נותר להראות ש־$P \cap S$, אם $s \in P \cap S$ אז $\frac{s}{1} \in {(S^{-1} A)}^\times$ וגם $\frac{s}{1} \in Q$ וזו סתירה לראשוניות $Q$.
\end{proof}
\begin{proposition}
	הפעולה $I \mapsto S^{-1} I$ שומרת הכלה, סכומים, חיתוכים, כפל ורדיקל.
\end{proposition}

\begin{definition}[מיקום של מודול]
	יהי $A$ חוג, $S \subset A$ כפלית, ויהי $M$ מודול מעל $A$.
	נגדיר יחס שקילות על $M \times S$ על־ידי,
	\[
		(x, s) \sim (y, t)
		\iff \exists r \in S,\ r t x = r s y
	\]
	ונסמן $\frac{x}{s} = {[ (x, s) ]}_\sim$.
	נגדיר,
	\[
		S^{-1} M
		= \{ \frac{x}{s} \mid x \in M, s \in S \}
	\]
	להיות המיקום של $M$.
\end{definition}
נגדיר חיבור על $S^{-1} M$ על־ידי,
\[
	\frac{x}{s} + \frac{y}{t}
	= \frac{t x + s y}{s t}
\]
ונגדיר כפל בסקלר מ־$S^{-1} A$ על־ידי,
\[
	\frac{a}{s} \cdot \frac{x}{t}
	= \frac{a x}{s t}
\]

	הראו כי הפעולות הללו מוגדרות היטב.

\begin{proposition}
	$S^{-1} A \otimes_A M \cong S^{-1} M$ מתקבל על־ידי,
	\[
		\frac{a}{s} \otimes x
		\mapsto \frac{a x}{s},
		\qquad
		\frac{x}{s}
		\mapsto \frac{1}{s} \otimes x
	\]
\end{proposition}
\begin{proof}
	ההעתקה $\frac{a}{s}, x \mapsto \frac{a x}{s}$ בילינארית, לכן היא מגדירה הומומורפיזם יחיד של $A$־מודולים,
	\[
		\varphi : S^{-1} A \otimes_A S^{-1} M
	\]
	כך ש־$\varphi(\frac{a}{s} \otimes x) = \frac{a x}{s}$. \\
	בכיוון ההפוך נגדיר את $\psi : S^{-1} M \to S^{-1} A \otimes M$ על־ידי $\psi(\frac{x}{s}) = \frac{1}{s} \otimes x$.
	$\psi$ מוגדר היטב שכן אם $\frac{x}{s} = \frac{y}{t}$ אז $r t x = r s y$ לאיזשהו $r \in S$ ולכן,
	\[
		\frac{1}{s} \otimes x
		= \frac{r t}{r t s} \otimes x
		= \frac{1}{r t s} \otimes r t x
		= \frac{r s}{r t s} \otimes y
		= \frac{1}{t} \otimes y
	\]
	קל לבדוק ש־$\psi$ הומומורפיזם וש־$\varphi, \psi$ הפוכים אחד לשני.
\end{proof}
\begin{definition}[מיקום של הומומורפיזם]
	אם $f : M \to N$ הומומורפיזם של $A$־מודולים אז נגדיר,
	\[
		S^{-1} f
		\in \Hom_{S^{-1} A}(S^{-1} M, S^{-1} N),
		\qquad
		(S^{-1} f)(\frac{x}{s})
		= \frac{f(x)}{s}
	\]
	אם נזהה $S^{-1} M = S^{-1} A \otimes M$ ו־$S^{-1} N = S^{-1} A \otimes N$ אז $S^{-1} f$ מזדהה עם $\id_{S^{-1} A} \otimes f$.
\end{definition}

\begin{proposition}
	תהי $0 \to N \to \xrightarrow{f} M \to \xrightarrow{g} P \to 0$ סדרה מדויקת קצרה. \\
	אז $0 \to S^{-1} N \xrightarrow{S^{-1} f} S^{-1} M \xrightarrow{S^{-1} g} S^{-1} P \to 0$ מדויקת.
\end{proposition}
\begin{proof}
	לפי הטענה הקודמת ודיוק מימין של טנזור מתקיים $S^{-1} N \to S^{-1} M \to S^{-1} P \to 0$ מדויקת.
	צריך להראות ש־$S^{-1} f$ חד־חד ערכית.
	נניח ש־$\frac{x}{s} \in \ker(S^{-1} f)$ עבור $s \in S, x \in N$, אז,
	\[
		0
		= S^{-1} f(\frac{x}{s})
		= \frac{f(x)}{s}
	\]
	ולכן קיים $r \in S$ כך ש־$f(r x) = r f(x) = 0$, $f$ חד־חד ערכית מדיוק הסדרה המקורית ולכן $r x = 0$ ו־$\frac{x}{s} = 0$.
\end{proof}
\begin{remark}
	אם נתונה פעולה על מודולים והומומורפיזמים ששומרת על הרכבה, סכום, הומומורפיזמים ודיוק של סדרות קצרות, אז היא שומרת דיוק של כל סדרה.
	בפרט אם $M_0 \xrightarrow{f_0} \cdots \xrightarrow{f_n} M_n$ מדויקת אז גם $S^{-1} M_0 \xrightarrow{S^{-1} f_1} \cdots \xrightarrow{S^{-1} f_n} S^{-1} M_n$ מדויקת.
	$0 \to \ker(f_{i + 1}) \hookrightarrow M_i \to f_{i + 1}(M_i) \to 0$.
\end{remark}
