\begin{exercise}
	הראו של־$2$ יש שורש ב־$\QQ_7$ ומצאו את שלוש הספרות שלו אחרי הנקודה.
\end{exercise}
\begin{solution}
	נגדיר את הפולינום $f \in \QQ_7[x]$,
	\[
		f(x)
		= x^2 - 2
	.\]
	ולכן הנגזרת הפורמלית שלו היא $f'(x) = 2x$, נסמן גם את $\bar{f}, \bar{f}' \in \FF_7$.
	מהלמה של הנזל אם קיים $w_0 \in \ZZ_7$ כך שמתקיים $\bar{f}(w_0) = 0$ וגם $\bar{f}'(w_0) \ne 0$,
	אז קיים מספר יחיד $w \in \ZZ_7$ כך ש־$f(w) = 0$ וגם $w \equiv w_0 \mod 7$.
	נסמן $w_0 = 3$ אז נקבל $f(w_0) = f(3) = 7$ ולכן $\bar{f}(w_0) = 0$ אבל $\bar{f}'(w_0) = 6 \ne 0$,
	לכן קיים $w \in \ZZ_7$ המקיים $w \equiv 3 \mod 7$.
	נסמן,
	\[
		w
		= \sum_{n = 0}^\infty a_n \cdot 7^n
		\tag{1}
	.\]
	עבור $0 \le a_n \le 6$ לכל $n \in \NN$. \\
	לכן מביצוע מודולו על המשוואה אנו מקבלים,
	\[
		w \equiv a_0 + 7 (\ldots) \mod 7
		\implies 3 \equiv a_0 \mod 7
	.\]
	ונסיק כי $a_0 = 3$ בלבד.
	נעבור לחישוב הספרה השנייה, שוב מ־$(1)$ נקבל על־ידי מודולו $7^2$,
	\begin{align*}
		w \equiv_{7^2} a_0 + 7 a_1 + 7^2 (\ldots)
		& \implies w^2 - 2 \equiv_{7^2} {(a_0 + 7 a_1)}^2 - 2 \\
		& \implies f(w) \equiv_{7^2} a_0^2 + 14 a_0 a_1 - 2 + 7^2 (\ldots) \\
		& \implies 0 \equiv_{7^2} 7 + 42 a_1 \\
		& \implies 0 \equiv_7 1 + 6 a_1
	.\end{align*}
	ונסיק $a_1 = 1$ בלבד. \\
	שוב מ־$(1)$ נקבל,
	\[
		w \equiv_{7^3} a_0 + a_1 7 + a_2 7^2
		\iff w \equiv_{7^3} 10 + 7^2 a_2
		\iff 0 \equiv_{7^3} 100 + 20 \cdot 7^2 a_2 - 2
		\iff 0 \equiv_7 2 + 20 a_2 \equiv_7 2 - a_2
	.\]
	ונסיק ש־$a_2 = 2$ בלבד.
	קיבלנו,
	\[
		w
		= {( \ldots\ 2\ 1\ 3 )}_7
	.\]
\end{solution}
\begin{exercise}
	מצאי בסיס מלכסנן קנונית של התבנית $q : \FF_5^3 \to \FF_5$ המוגדרת על־ידי,
	\[
		q\begin{pmatrix} x_1 \\ x_2 \\ x_3 \end{pmatrix}
		= -2 x_1^2 + 2 x_2^2 + 2 x_2 x_3 - x_3^2
	.\]
	עבור קבוצת הנציגים $r = \{1, 2\}$.
\end{exercise}
\begin{solution}
	נבחין כי מתקיים,
	\[
		{[g_q]}_E
		= \begin{pmatrix}
			-2 & 0 & 0 \\
			0 & 2 & 1 \\
			0 & 1 & -1
		\end{pmatrix}
	.\]
	מובטח לנו שקיים וקטור שרק שתי הקורדינטות הראשונות שלו שונות מאפס ושעבורו $q(v) = 1$, נמצא כזה עם טבלה.
	\begin{otherlanguage}{english}
		\begin{center}
			\begin{tabular}{c | c c c c c c}
				$x$ & $-2$ & $-1$ & $0$ & $1$ & $2$ \\
				\hline
				$x^2$ & $-1$ & $1$ & $0$ & $1$ & $-1$ \\
				\hline
				$-2x^2$ & $2$ & $-2$ & $0$ & $-2$ & $2$ \\
				\hline
				$1 - 2x^2$ & $-2$ & $-1$ & $1$ & $-1$ & $-2$ \\
			\end{tabular}
		\end{center}
	\end{otherlanguage}
	נבחין שיש תוצאות שוות בשורות השלישית והרביעית, הן מסמלות שקיימת בחירה $-2x_1^2 = 1 - 2x_2^2$,
	אז נגדיר את,
	\[
		u_1
		= \begin{pmatrix} 1 \\ 2 \\ 0 \end{pmatrix}
	.\]
	וזהו למעשה הווקטור הראשון לבסיס שלנו כי מתקבל $q(u_1) = 1$ ולכן לא צריך לשנות אותו בכלל.
	עתה נעבור לחפיפה של ${[g_q]}_{(u_1, e_2, e_3)}$
	\[
		\begin{pmatrix}
			1 & -1 & 2 \\
			-1 & 2 & 1 \\
			2 & 1 & -1 \\
			\hline
			1 & 0 & 0 \\
			2 & 1 & 0 \\
			0 & 0 & 1
		\end{pmatrix}
		\xrightarrow[C_2 \to C_2 + C_1]{R_2 \to R_2 + R_1}
		\begin{pmatrix}
			1 & 0 & 2 \\
			0 & 1 & -2 \\
			2 & -2 & -1 \\
			\hline
			1 & 1 & 0 \\
			2 & -2 & 0 \\
			0 & 0 & 1
		\end{pmatrix}
		\xrightarrow[C_3 \to C_3 - 2 C_1]{R_3 \to R_3 - 2 R_1}
		\begin{pmatrix}
			1 & 0 & 0 \\
			0 & 1 & -2 \\
			0 & -2 & 0 \\
			\hline
			1 & 1 & -2 \\
			2 & -2 & 1 \\
			0 & 0 & 1
		\end{pmatrix}
		\xrightarrow[C_3 \to C_3 + 2 C_2]{R_3 \to R_3 + 2 R_2}
		\begin{pmatrix}
			1 & 0 & 0 \\
			0 & 1 & 0 \\
			0 & 0 & 1 \\
			\hline
			1 & 1 & 0 \\
			2 & -2 & 2 \\
			0 & 0 & 1
		\end{pmatrix}
	\]
	נסמן,
	\[
		D
		= \begin{pmatrix}
			1 & 1 & 0 \\
			2 & -2 & 2 \\
			0 & 0 & 1
		\end{pmatrix},
		\Bb
		= \left( u_1,
		\begin{pmatrix} 1 \\ -2 \\ 0 \end{pmatrix},
		\begin{pmatrix} 0 \\ 2 \\ 1 \end{pmatrix} \right)
	.\]
	אז לפי החפיפה נקבל,
	\[
		{[g_q]}_\Bb
		= D^t {[g_q]}_E D
		= I_3
	.\]
	ובפרט מצאנו שהצורה הקנונית של $q$ היא,
	\[
		q' \begin{pmatrix} x_1 \\ x_2 \\ x_3 \end{pmatrix}
		= x_1^2 + x_2^2 + x_3^2
	.\]
\end{solution}

\begin{enumerate}
	\item
		תהי $X \subseteq \RR^n$ יריעה חלקה.
		\begin{enumerate}
			\item נגדיר את הקבוצה $\Bb$ בדרך הבאה:
				לכל $x \in X$ ופרמטריזציה $\varphi : U \to X$ כך ש־$x \in U \subseteq \HH^d$ אז לכל $W \subseteq U$ פתוחה מתקיים $\varphi(W) \in \Bb$. \\
				הראי כי $\Bb$ מהווה בסיס לטופולוגיה, נסמנה $\Tt_X$, תארי את הטופולוגיה במילים.
			\item נגדיר את הטופולוגיה $\Tt_X'$ כטופולוגיה המצומצמת של $\RR^n$,
				הראי ש־$\Tt_X = \Tt_X'$ והסיקי שהטופולוגיה מקיימת את אקסיומת המנייה השנייה.
			\item נניח ש־$X$ היא יריעה קומפקטית וקשירה מסילתית, הראי כי גם $(X, \Tt_X)$ קומפקטית וקשירה מסילתית (בטופולוגיה במקום במובן המטרי).
			\item תהי $I$ קבוצת אינדקסים ויהי ${\{ W_i \}}_{i \in I} \subseteq X$ כיסוי פתוחות של $X$ כך ש־$\varphi_i : U_i \to W_i$ דיפאומורפיזם עבור $U_i \subseteq \HH^d$.
				הראי כי בלי הגבלת הכלליות $I$ סופית ושקיים $C \in \RR$ כך ש־$C > 0$ וגם $\operatorname{diam}(U_i) < C$ לכל $i$.
			\item נניח בלי הגבלת הכלליות ש־$I = [k]$ ונגדיר את המרחב,
				\[
					\Omega
					= \bigcup_{i = 1}^k (\{ (i \cdot C, 0, \ldots, 0) \} + U_i)
				.\]
				הראי כי $\Omega$ לא קשירה, וכי יש לה בדיוק $k$ מחלקות קשירות.
			\item נגדיר את ההעתקה $p : \Omega \to X$ באופן הבא:
				אם $x \in \{ (i \cdot C, 0, \ldots, 0) \} + U_i$ עבור $i \le k$ כלשהו אז,
				\[
					p(x)
					= \varphi_i(x - (i C, 0, \ldots, 0))
				.\]
				הראי כי $p$ העתקת כיסוי.
			\item נניח ש־$X$ הוא שפתו של גליל, כלומר,
				\[
					X = S^1 \times (0, H)
					\subseteq \RR^3
				.\]
				עד כדי הזזה.
				הראי כי קיים דיפאומורפיזם מהקבוצה $B(0, 2) \setminus \overline{B}(0, 1) \subseteq \RR^2$ ל־$X$ והסיקי כי מתקיים,
				\[
					\pi_1(X, x_0)
					\cong \ZZ
				.\]
				כלומר החבורה היסודית של הגליל היא החבורה היסודית של המעגל.
			\item יהי גליל אינסופי, כלומר,
				\[
					X = S^1 \times \RR
				.\]
				הראי כי התוצאה מהסעיף הקודם נכונה גם הפעם על־ידי שימוש בכלים טופולוגיים בלבד.
		\end{enumerate}
	\item חישובי גובה. \\
		אם נתונה לנו פרמטריזציה של יריעה כלשהי $(X, \varphi : U \to X)$ אז אנו יכולים להגדיר עליה פונקציית גובה,
		כלומר פונקציה שמתארת לנו עבור כל נקודה מה ה''גובה'' שלה. אם לדוגמה נבחר את היריעה $X = [0, 1]$, כלומר קטע היחידה הסגור, אז נוכל לתאר את הגובה שלה בכמה דרכים,
		לדוגמה,
		\[
			h_1(t)
			= 0,
			\qquad
			h_2(t) = t
		.\]
		הפונקציה הראשונה מתארת את הגובה כאשר המוט שבחרנו שוכב על הרצפה, והפונקציה השנייה מתארת את הגובה כאשר המוט נעוץ בקרקע.
		באופן כללי נבחין כי לא קיימת פונקציית גובה יחידה ליריעה פרמטרית, אבל בתרגיל זה הפונקציה תהיה באופן גלובלי ההטלה על ציר $z$, כלומר הפונקציה,
		\[
			h(x, y, z)
			= z
		.\]
		והיא גורמת לזה שהפרמטריזציה תגדיר את הגובה בעצם הדרך שבה היא נתונה.
		בכל סעיף תופיע יריעה ועלייך לחשב את הגובה הממוצע של הנקודות, כלומר האינטגרל המנורמל של $h$ על היריעה.
		\begin{enumerate}
			\item מוט יושב,
				\[
					\varphi : [0, 1] \to \RR^3,
					\quad
					\varphi(t)
					= (t, 0, 0)
				.\]
			\item מוט עומד,
				\[
					\varphi : [0, 1] \to \RR^3,
					\quad
					\varphi(t)
					= (0, 0, t)
				.\]
			\item ריבוע יושב,
				\[
					\varphi : {[0, 1]}^2 \to \RR^3,
					\quad
					\varphi(t, s)
					= (t, s, 0)
				.\]
			\item ריבוע עומד,
				\[
					\varphi : {[0, 1]}^2 \to \RR^3,
					\quad
					\varphi(t, s)
					= (0, t, s)
				.\]
			\item גליל ברדיוס $R$ וגובה $H$,
				\[
					\varphi : [0, 2 \pi] \times [0, H],
					\quad
					\varphi(s, t)
					= (R \cos s, R \sin s, t)
				.\]
		\end{enumerate}
	\item בכל סעיף נגדיר את הקבוצה $M \subseteq \RR^3$, בדקי אם היא יריעה או לא.
		\begin{enumerate}
			\item $M = \{ (x, y, z) \in \RR^3 \mid x^2 z + y^2 z = z^2 \}$
			\item $M = \{ (x, y, z) \in \RR^3 \mid x^2 z + y^2 z = z^2 + 1 \}$
			\item $M = \{ (x, y, z) \in \RR^3 \mid x^2 = y^2 + z \}$
			\item $M = \{ (x, y, z) \in \RR^3 \mid x^2 + y^2 = \cos^2 z \}$
			\item $M = \{ (x, y, z) \in \RR^3 \mid |x| + |y| + |z| = 1 \}$
			\item $M = \{ (x, y, z) \in \RR^3 \mid zx^{2}+zy^{2}+z^{3}-4z^{2}+3z=0 \}$
			\item $M = \{ (x, y, z) \in \RR^3 \mid x y = 0, |z| \le 1 \}$
			\item $M = \{ (x, y, z) \in \RR^3 \mid (x, y, z) \ne 0 \}$
			\item $M = \{ (x, y, z) \in \RR^3 \mid x^{2}+z^{2}=e^{y} \}$
		\end{enumerate}
	\item תהי פונקציה $f : \RR \to \RR$ חלקה וחיובית (כלומר $f(x) > 0$ לכל $x$).
		נגדיר את הקבוצה,
		\[
			M
			= \{ (x, y, z) \in \RR^3 \mid x^2 + y^2 = {(f(z))}^2 \}
		.\]
		הוכיחי כי $M$ היא יריעה $2$־מימדית במרחב.
\end{enumerate}
